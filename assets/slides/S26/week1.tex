\section{continuity and integration}


\begin{frame}{continuity and $C([a,b])$}

\begin{definition}[continuity]
\begin{itemize}
\item a function $f:[a,b] \to \RR$ is \emph{continuous at} $x\in[a,b]$ if for all $\eps>0$ there is $\delta>0$ so that
  $$|y-x|<\delta \text{ and } y\in[a,b] \quad \implies \quad |f(y)-f(x)|<\eps$$
\item equivalently: \quad $\ds f(x) = \lim_{y\to x} f(y)$ \quad (where the limit exists!)
\item a function $f:[a,b] \to \RR$ is \emph{continuous} if it is continuous at each $x\in[a,b]$
\end{itemize}
\end{definition}

\begin{frame}{continuity and $C([a,b])$}

\begin{definition}

\begin{frame}{$C([a,b])$ is vector space}

\begin{itemize}
\item what is the definition of a \emph{vector space}?
\item ``linear space'' is a synonym for vector space
\end{itemize}

\begin{theorem}
$V=C([a,b])$ is a (real) vector space
\end{theorem}

\begin{proof}
If $f,g \in V$ then $h(x) = f(x)+g(x)=g(x)+f(x)$ is a continuous function, so $V$ is closed under addition.  If $\alpha\in \RR$ then $k(x)=\alpha f(x)$ is a continuous function, so $V$ is closed under scalar multiplication.   The zero function is a continuous function, and $f(x)+0=f(x)$, so $V$ has an additive identity.  Also $1 f(x) = f(x)$, so there is scalar identity.  And distribution rules hold: $\alpha(f+g) = \alpha f + \alpha g$, $\alpha(\beta f) = (\alpha \beta) f$.
\end{proof}

\vspace{30mm}
\end{frame}


\begin{frame}{derivatives \dots may not exist for continuous functions}

\begin{definition}[ordinary derivative]
given $f:[a,b] \to \RR$ and $c\in (a,b)$ define \quad $\ds f'(c) = \lim_{h\to 0} \frac{f(c+h) - f(c)}{h}$

to be the \emph{derivative of $f$ at $c$}, but only \emph{if} the limit exists
\end{definition}

\begin{itemize}
\item for $f(x) = x(1-x) \in C([0,1])$ we have $f'(c) = 1 - 2 c$ for $c\in (0,1)$
\item for $f(x) = |1 - 2 x| \in C([0,1])$,  the limit to define $f'(0.5)$ does not exist
\end{itemize}

\begin{frame}{definite integral}

\begin{definition}[Riemann's integral]
for meshes $a = x_0 < x_1 < \dots < x_j < \dots < x_n = b$, and evaluation points $x_j^* \in [x_{j-1},x_j]$, define \quad $\ds \int_a^b f(x)\,dx = \lim_{n\to\infty} \sum_{j=1}^n f(x_j^*)\,(x_j - x_{j-1})$

as the \emph{Riemann integral} of $f$ over $[a,b]$, but only \emph{if} the limit exists
\end{definition}

\begin{itemize}
\item the limit process for an integral is more subtle than the limit process for continuity or derivatives
    \begin{itemize}
    \item[$\circ$] this limit \emph{does not} exist for some (non-continuous) functions
    \end{itemize}
\item Riemann's is not the only definition of an integral
    \begin{itemize}
    \item[$\circ$] we will get to Lebesgue's integral later
    \item[$\circ$] it agrees with Riemann's when they are both defined
    \item[$\circ$] we will use Lebesgue integration routinely, but only lightly cover its details
    \end{itemize}
\end{itemize}

\begin{theorem}
if $f:[a,b] \to \RR$ is continuous then $\ds \int_a^b f(x)\,dx \in \RR$ is well-defined
\end{theorem}
\end{frame}

\begin{frame}{linear functionals on $C([0,1])$}

\begin{definition}[linear functional]
a function $\ell:C([0,1]) \to \RR$ is a \emph{linear functional} if
	$$\ell[\alpha f + \beta g] = \alpha \ell[f] + \beta \ell[g]$$
for all $\alpha,\beta \in\RR$ and $f,g\in C([0,1])$
\end{definition}

\section{numerical integration}

\begin{frame}{numerical integration rules}

\begin{itemize}
\item \textbf{trapezoid rule} for $m$ equal intervals on $[0,1]$:
     $$\int_0^1 f(x)\,dx \approx T_m(f) = \frac{h}{2} \Big(f(x_0) + 2 f(x_1) + \dots + 2 f(x_{m-1}) + f(x_m)\Big)$$
where $h=1/m$ and $x_j = j h$
\item \textbf{Simpson's rule} for \emph{even} $m$ intervals, and same $h$, $x_j$:
\begin{align*}
\int_0^1 f(x)\,dx \approx S_m(f) = \frac{h}{3} &\Big(f(x_0) + 4 f(x_1) + 2 f(x_2) + 4 f(x_3) + 2 f(x_4) \\
&\qquad + \dots + 4 f(x_{m-1}) + f(x_m)\Big)
\end{align*}
\item \dots there are many more rules, and many are better, but they have the same flavor \dots
\end{itemize}
\end{frame}

\begin{frame}{linear functionals in finite dimensions}

\begin{itemize}
\item suppose we have a real, finite-dimensional vector space $V$
\item suppose we have a basis for $V$: \quad $\mathcal{B} = \{b_1,\dots,b_m\} \subset V$
\item write $f\in V$ in this basis: \quad for $f_j\in \RR$,
	$$f = f_1 b_1 + \dots f_m b_m$$
\item the coefficients are usually written in a column vector:
  $$(f)_{\mathcal{B}} = \begin{bmatrix} f_1 \\ f_2 \\ \vdots \\ f_m \end{bmatrix}$$
\end{itemize}
\end{frame}


\begin{frame}{linear functionals in finite dimensions}

\begin{theorem}
if $V$ is a real, finite-dimensional vector space with a basis $\mathcal{B} = \{b_1,\dots,b_m\}$, and if $f = f_1 b_1 + \dots f_m b_m$ in this basis, then for any linear functional $\ell:V \to \RR$ there are values $\ell_j$, $j=1,\dots,m$, so that

\vspace{-10mm}
	$$\ell[f] = \begin{bmatrix} \ell_1 & \ell_2 & \dots & \ell_m \end{bmatrix}\begin{matrix} \vspace{8mm} \\ \begin{bmatrix} f_1 \\ f_2 \\ \vdots \\ f_m \end{bmatrix} \end{matrix} = \sum_{j=1}^m \ell_j f_j$$

\smallskip
\end{theorem}

\begin{itemize}
\item in other words, we represent linear functions by \emph{row vectors} when the vectors themselves are \emph{column vectors}
\item \dots so that $\ell[f]$ is computed by usual rules of matrix-matrix multiplication
\end{itemize}
\end{frame}
