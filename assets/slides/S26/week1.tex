\begin{frame}{continuity and $C([a,b])$}

\begin{itemize}

\begin{definition}[continuity]
\begin{itemize}
\item a function $f:[a,b] \to \RR$ is \emph{continuous at} $x\in[a,b]$ if for all $\eps>0$ there is $\delta>0$ so that
  $$|y-x|<\delta \text{ and } y\in[a,b] \quad \implies \quad |f(y)-f(x)|<\eps$$
\item a function $f:[a,b] \to \RR$ is \emph{continuous} if it is continuous at each $x\in[a,b]$
\end{itemize}
\end{definition}

\begin{frame}{$C([0,1])$ is vector space}

\begin{theorem}
$V=C([0,1])$ is a (real) vector space
\end{theorem}

\begin{proof}
If $f,g \in V$ then $h(x) = f(x)+g(x)=g(x)+f(x)$ is continuous,  so $V$ is closed under addition.  If $\alpha\in \RR$ then $\alpha f(x)$ is continuous, so $V$ is closed under scalar multiplication.   The zero function is a continuous function, and $f(x)+0=f(x)$, so $V$ has an additive identity.  Also $1 f(x) = f(x)$, so the scalar identity works.  And distribution rules are true: $\alpha(f+g) = \alpha f + \alpha g$, $\alpha(\beta f) = (\alpha \beta) f$.
\end{proof}

\begin{itemize}
\item ``linear space'' is a synonym for vector space
\item wait \dots what is the definition of a \emph{vector space}?
\end{itemize}

\begin{frame}{definite integral}

\begin{definition}[Riemann's integral]
for a mesh $a = x_0 < x_1 < \dots < x_j < \dots < x_n = b$, and evaluation points $x_j^* \in [x_{j-1},x_j]$, define
  $$\int_a^b f(x)\,dx = \lim_{n\to\infty} \sum_{j=1}^n f(x_j^*)\,(x_j - x_{j-1})$$ 
\end{definition}

\begin{itemize}
\item this is \emph{not} the only definition of an integral
\item the limit process for an integral is more subtle than the limit process for continuity or derivatives
\item we will get to Lebesgue's integral later; we will use it a lot, but only lightly cover its details
\end{itemize}

\begin{theorem}
if $f:[a,b] \to \RR$ is continuous then $\int_a^b f(x)\,dx \in \RR$ is well-defined
\end{theorem}
\end{frame}

	$$\ell[\alpha f + \beta g] = \alpha \ell[f] + \beta \ell[g]$$

\begin{frame}{questions about linear functionals}

\begin{block}{regularity question}
we don't need $f$ and $w$ to be continuous for this to be well-defined:
	$$\ell[f] = \int_0^1 f(x) w(x)\,dx$$
how general can these functions be?  can they be discontinuous? completely arbitrary?
\end{block}

\begin{block}{recovery question}
suppose we think $\ell$ has a formula $\ell[f] = \int_0^1 f(x) w(x)\,dx$, but we don't know $w(x)$.  can we recover $w(x)$ from the values $\ell[f]$ for various $f$?
\end{block}

\begin{block}{conceptual question}
consider the space $V'$ of all linear functionals on $V=C([0,1])$.  is it a vector space? (yes!)  how do we \emph{concretelly} understand this abstractly-defined space?  is each $\ell \in V'$ represented by a $w(x)$?

\section{numerical integration}

\begin{frame}{numerical integration rules}

\begin{itemize}
\item \textbf{trapezoid rule} for $m$ equal intervals on $[0,1]$:
     $$\int_0^1 f(x)\,dx \approx T_m(f) = \frac{h}{2} \Big(f(x_0) + 2 f(x_1) + \dots + 2 f(x_{m-1}) + f(x_m)\Big)$$
where $h=1/m$ and $x_j = j h$
\item \textbf{Simpson's rule} for \emph{even} $m$ intervals, and same $h$, $x_j$:
\begin{align*}
\int_0^1 f(x)\,dx \approx S_m(f) = \frac{h}{3} &\Big(f(x_0) + 4 f(x_1) + 2 f(x_2) + 4 f(x_3) + 2 f(x_4) \\
&\qquad + \dots + 4 f(x_{m-1}) + f(x_m)\Big)
\end{align*}
\item \dots there are many more rules, and many are better, but they have the same flavor \dots
\end{itemize}

\begin{frame}{linear functionals in finite dimensions}

\begin{itemize}
\item suppose we have a real, finite-dimensional vector space $V$
\item suppose we have a basis for $V$: \quad $\mathcal{B} = \{b_1,\dots,b_m\} \subset V$
\item write $f\in V$ in this basis: \quad for $f_j\in \RR$,
	$$f = f_1 b_1 + \dots f_m b_m$$
\item the coefficients are usually written in a column vector:
  $$(f)_{\mathcal{B}} = \begin{bmatrix} f_1 \\ f_2 \\ \vdots \\ f_m \end{bmatrix}$$
\end{itemize}
\end{frame}


\begin{frame}{linear functionals in finite dimensions}

\begin{theorem}
if $\ell:V \to \RR$ is a linear functional then there are values $\ell_j$, $j=1,\dots,m$, so that

\vspace{-10mm}
	$$\ell[f] = \begin{bmatrix} \ell_1 & \ell_2 & \dots & \ell_m \end{bmatrix}\begin{matrix} \vspace{8mm} \\ \begin{bmatrix} f_1 \\ f_2 \\ \vdots \\ f_m \end{bmatrix} \end{matrix} = \sum_{j=1}^m \ell_j f_j$$

\smallskip
\end{theorem}

\begin{itemize}
\item in other words, we usually represent linear functions by \emph{row vectors} when we are representing the vectors themselves by \emph{column vectors}
\item then $\ell[f]$ is computed by usual rules of matrix-matrix multiplication
\end{itemize}
\end{frame}
