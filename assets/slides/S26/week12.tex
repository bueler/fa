\begin{frame}{the Laplacian}

\begin{itemize}
\item given $f$, finding $u$ from the Poisson equation $- \grad^2 u = f$ is a problem of inverting the operator $-\grad^2$

\begin{definition}[Laplacian operator]
for $\Omega\subset\RR^d$ open and $u:\Omega\to \RR$, the \emph{Laplacian} of $u$ is
\begin{equation*}
  \grad^2 u = \Div(\grad u) = \frac{\partial^2 u}{\partial x_1^2} +\dots + \frac{\partial^2 u}{\partial x_d^2}
\end{equation*}
\end{definition}

\item $\grad^2 u \stackrel{d=1}{=} u'', \quad \grad^2 u \stackrel{d=2}{=} u_{xx}+u_{yy}, \quad \grad^2 u \stackrel{d=3}{=} u_{xx}+u_{yy}+u_{zz}$
\end{itemize}

\begin{definition}[Laplace's equation $=$ potential equation]
\begin{equation*}
  \grad^2 u = 0,
\end{equation*}
and a solution of this equation is called \emph{harmonic}
\end{definition}
\end{frame}


\begin{frame}{tool: divergence theorem}

\begin{itemize}
\item $\grad^2=\Div \grad$ usually appears in models because a flux is proportional to a gradient, and then a divergence connects interior behavior to a boundary integral of the flux
\item we will need the divergence theorem from the \href{https://bueler.github.io/fa/assets/slides/S26/week10.pdf}{weeks 10\&11 slides}:
   $$\int_\Omega \Div \bV\,dm = \int_{\partial\Omega} \bV\cdot \bn\,ds,$$

   \begin{itemize}
   \item[$\circ$] there is an $\RR^d$ version of this theorem for any $d$
   \end{itemize}
\item when using this, $\bV$ will be the gradient of a scalar function
\item recall the boundary integral is the total flux of $\bV$ outward through the boundary
\end{itemize}
\end{frame}


\begin{frame}{boundary-value problems for the Poisson equation}

\begin{itemize}
\item in the absence of any boundary conditions there is an infinite-dimensional space of solutions to $-\grad^2 u = f$ on $\Omega$
\item if $f=0$ then this is the space of all \emph{harmonic functions} on $\Omega$, an $\infty$-dimensional subspace of all smooth functions:
    $$\left\{w\,:\,\grad^2 w= 0\right\} \subset C^2(\Omega)$$

   \begin{itemize}
   \item[$\circ$] complex analysis is, essentially, the study of this space
   \end{itemize}
\item but we want to build models which make \emph{one} prediction, so we will focus on \emph{boundary-value problems} that pick-out one solution $u$
\end{itemize}

\begin{block}{Dirichlet problem for the Poisson equation}
for $f\in L^2(\Omega)$ and $g_D\in L^2(\partial\Omega)$, find $u:\Omega\to \RR$ so that
\begin{align*}
- \grad^2 u &= f \quad \text{ on } \Omega \\
u &= g_D \quad \text{ on } \partial \Omega
\end{align*}
\end{block}
\end{frame}


\begin{frame}{boundary-value problems for the Poisson equation}

\begin{itemize}
\item instead of $u$ itself, the gradient $\grad u$ could be set to known values along the boundary, which makes sense because the gradient is often proportional to a physical flux
\end{itemize}

\begin{block}{Neumann problem for the Poisson equation}
for $f\in L^2(\Omega)$ and $g_N\in L^2(\partial\Omega)$, find $u:\Omega\to \RR$ so that
\begin{align*}
- \grad^2 u &= f \quad \text{ on } \Omega \\
\grad u\cdot \bn &= g_N \quad \text{ on } \partial \Omega
\end{align*}
where $\bn$ is the outward unit normal vector field along $\partial\Omega$
\end{block}

\begin{itemize}
\item other notation for the (scalar) directional derivative:
	$$\grad u\cdot \bn = \frac{\partial u}{\partial n} = u_n$$
\item one could have Dirichlet conditions on part of the boundary and Neumann on another part, so that the boundary is a disjoint union:
	$$\partial \Omega = \partial_D \Omega \cup \partial_N \Omega$$
\end{itemize}
\end{frame}


\begin{frame}{applications}

\begin{itemize}
\item the Poisson equation can be used to model heat conduction, electrostatic potential, elastic membranes, equilibrium distributions for random walks, or many other physical phenomena
\end{itemize}

\begin{block}{application: temperature of a conductive solid}
\begin{itemize}
\small
\item Fourier's law for heat conduction in solids says that the heat flux is $\bq = -k \grad u$, where $u$ is the temperature, $\bq$ is the heat flux, and $k>0$ is the conductivity
\item consider a solid with a heat capacity $c>0$ (heat gained by given increase in temperature) and a mass density $\rho>0$ (mass per unit volume)
\item given a heat source $f$ within the domain $\Omega$, conservation of energy says
    $$c\rho\, \frac{\partial u}{\partial t} = - \Div\bq + f \quad \iff \quad \frac{d}{dt}\left(\int_\Omega c\rho\, u \,dx\right) = - \int_{\partial\Omega} \bq\cdot\bn\,ds + \int_\Omega f \,dx$$
\item assume steady state, so $\partial/\partial t=d/d t=0$
\item if $k$ is constant then $- \Div\bq = - \Div(-k \grad u) = k \grad^2 u$
\item we derive Poisson's equation, the (steady) \emph{heat} or \emph{diffusion equation}
    $$0 = k \grad^2 u + f \qquad \text{ within } \Omega$$
\end{itemize}
\end{block}
\end{frame}


\begin{frame}{applications}

\begin{block}{application: deflection of an elastic membrane}
\begin{itemize}
\small
\item suppose $z=u(x,y)$ is a vertical displacement of an elastic membrane over a domain $\Omega\subset \RR^2$
\item suppose that along $\partial \Omega$ we have put a rigid frame with height $z=g(x,y)$
\item suppose that the slopes along the graph of $g$ are small
\item suppose that the membrane is loaded by a force $f$ (per unit area)
\item group the elastic properties of the membrane into a constant $\kappa>0$
\item then the vertical displacement satisfies Poisson's equation
    $$- \kappa \grad^2 u = f \qquad \text{ within } \Omega$$
subject to $u=g$ along $\partial \Omega$
\end{itemize}
\end{block}
\end{frame}

\section{the Sobolev spaces $H^k(\Omega)$}

\begin{frame}{first Sobolev space}

\begin{itemize}
\item given a bounded open set $\Omega\subset \RR^d$
\item recall the definition of (real-valued) $L^2(\Omega)$
\item recall that we say $f$ has a weak (partial) derivative $g=\partial f/\partial x_k$ if $g\in L^1(\Omega)$ so that
    $$\int_\Omega g\, \varphi\,dm = - \int_\Omega f\, \frac{\partial\varphi}{\partial x_k}\,dm \qquad \text{for all } \varphi \in C_c^\infty(\Omega)$$
\item note that if $\Omega$ is bounded then $L^2(\Omega) \subset L^1(\Omega)$
\end{itemize}

\begin{definition}[Sobolev space $H^1(\Omega)$]
    $$H^1(\Omega) = \left\{f\in L^2(\Omega) \,\Big|\, \begin{matrix}  \text{the weak partial derivatives } \frac{\partial f}{\partial x_k} \text{ exist} \\ \text{and are in } L^2(\Omega) \text{ for } k=1,\dots,d \end{matrix} \right\}$$
\end{definition}

\begin{itemize}
\item ``$H$\,'' is for Hilbert
\end{itemize}

\end{frame}


\begin{frame}{more Sobolev spaces}

\begin{itemize}
\item recall our multi-index derivative notation from the \href{https://bueler.github.io/fa/assets/slides/S26/week10.pdf}{weeks 10\&11 slides}
\end{itemize}

\begin{definition}[weak derivative]
\small
if $f\in \Lloc(\Omega)$, and if $g\in \Lloc(\Omega)$ satisfies
    $$\int_\Omega g\, \varphi\,dm = (-1)^{|\alpha|} \int_\Omega f\, D^\alpha\varphi\,dm \qquad \text{for all } \varphi \in C_c^\infty(\Omega)$$
then we say $g$ is the \emph{$\alpha$th weak derivative} of $f$, and we write $D^\alpha f = g$ a.e.
\end{definition}

\begin{definition}[Sobolev space $H^k(\Omega)$]
    $$H^k(\Omega) = \left\{f\in L^2(\Omega) \,\Big|\, \begin{matrix}  \text{the weak derivatives } D^\alpha f \text{ exist} \\ \text{and are in } L^2(\Omega) \text{ for all } |\alpha|\le k\end{matrix} \right\}$$
\end{definition}

\begin{itemize}
\item we may write $H^0(\Omega) = L^2(\Omega)$
\item other books use notation $H^k(\Omega) = W^{k,2}(\Omega)$; the ``$2$'' indicates that the weak derivatives are in $L^2(\Omega)$
\item $W^{k,p}(\Omega)$ is the general case, with weak derivatives in $L^p(\Omega)$ up to order $k$
\end{itemize}
\end{frame}
