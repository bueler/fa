\section{the Poisson equation: $- \grad^2 u = f$}

\begin{frame}{the Poisson equation}

\begin{itemize}
\item let $\Omega \subset \RR^d$ be open; denote points by $x=(x_1,\dots,x_d)\in\Omega$

\begin{block}{Poisson equation}  find $u=u(x)$ so that
   $$- \grad^2 u = f \qquad \text{ on } \Omega$$
\end{block}

\item next slide defines ``$\grad^2$'' \dots
\item picture:
\end{itemize}

\vspace{35mm}
\end{frame}


\begin{frame}{the Laplacian}

\begin{itemize}
\item given $f$, finding $u$ from the Poisson equation $- \grad^2 u = f$ is a problem of inverting the operator $-\grad^2$

\begin{definition}[Laplacian operator]
for $\Omega\subset\RR^d$ open and $u:\Omega\to \RR$, the \emph{Laplacian} of $u$ is
\begin{equation*}
  \grad^2 u = \Div(\grad u) = \frac{\partial^2 u}{\partial x_1^2} +\dots + \frac{\partial^2 u}{\partial x_d^2}
\end{equation*}
\end{definition}

\item in each dimension:
\begin{align*}
d=1: && \grad^2 u &= u'' \\
d=2: && \grad^2 u &= u_{xx}+u_{yy} \\
d=3: && \grad^2 u &= u_{xx}+u_{yy}+u_{zz}
\end{align*}
\end{itemize}
\end{frame}


\begin{frame}{tool: divergence theorem}

\begin{itemize}
\item $\grad^2=\Div \grad$ usually appears in models because a flux is proportional to a gradient, and then a divergence connects interior behavior to a boundary integral of the flux
\item we will need the divergence theorem from the \href{https://bueler.github.io/fa/assets/slides/S26/week10.pdf}{weeks 10\&11 slides}:
   $$\int_\Omega \Div \bV\,dm = \int_{\partial\Omega} \bV\cdot \bn\,ds,$$

   \begin{itemize}
   \item[$\circ$] recall that this theorem applies to $\Omega\subset \RR^d$ if $\partial\Omega$ is Lipschitz or $C^1$
   \end{itemize}
\item when using this, $\bV$ will be the gradient of a scalar function
\item recall the boundary integral is the total flux of $\bV$ outward through the boundary
\end{itemize}
\end{frame}


\begin{frame}{Laplace's equation, and harmonic functions}

\begin{itemize}
\item in the absence of any boundary conditions there is an infinite-dimensional space of solutions to $-\grad^2 u = f$ on $\Omega$

\begin{definition}[Laplace's equation $=$ potential equation]
\begin{equation*}
  \grad^2 u = 0
\end{equation*}
a solution of this $f=0$ equation is called \emph{harmonic}
\end{definition}

\item the set of harmonic functions on $\Omega$ is an $\infty$-dimensional vector space of smooth functions:
    $$\left\{w\,:\,\grad^2 w= 0\right\} \subset C^\infty(\Omega)$$
\item complex analysis is, essentially, the study of this space
\end{itemize}
\end{frame}


\begin{frame}{boundary-value problems for the Poisson equation}

\begin{itemize}
\item however, we want to build models which make \emph{one} prediction
\item the Poisson equation can be used to model heat conduction, electrostatic potential, elastic membranes, equilibrium distributions for random walks, or many other physical phenomena
\item we will focus on \emph{boundary-value problems} that pick-out one solution $u$
\end{itemize}

\begin{block}{Dirichlet problem for the Poisson equation}
for $f\in L^2(\Omega)$ and $g_D\in L^2(\partial\Omega)$, find $u:\Omega\to \RR$ so that
\begin{align*}
- \grad^2 u &= f \quad \text{ on } \Omega \\
u &= g_D \quad \text{ on } \partial \Omega
\end{align*}
\end{block}
\end{frame}


\begin{frame}{boundary-value problems for the Poisson equation}

\begin{itemize}
\item instead of $u$ itself, the gradient $\grad u$ could be set to known values along the boundary
\item in models, the gradient is often proportional to a physical flux
\end{itemize}

\begin{block}{Neumann problem for the Poisson equation}
for $f\in L^2(\Omega)$ and $g_N\in L^2(\partial\Omega)$, find $u:\Omega\to \RR$ so that
\begin{align*}
- \grad^2 u &= f \quad \text{ on } \Omega \\
\grad u\cdot \bn &= g_N \quad \text{ on } \partial \Omega
\end{align*}
where $\bn$ is the outward unit normal vector field along $\partial\Omega$
\end{block}

\begin{itemize}
\item other notation for the (scalar) directional derivative:
	$$\grad u\cdot \bn = \frac{\partial u}{\partial n} = u_n$$
\item one could have Dirichlet conditions on part of the boundary and Neumann on another part, as a disjoint union:
	$$\partial \Omega = \partial_D \Omega \cup \partial_N \Omega$$
\end{itemize}
\end{frame}


\begin{frame}{applications}

\begin{block}{application: temperature of a conductive solid}
\begin{itemize}
\small
\item Fourier's law for heat conduction in solids says that the heat flux is
    $$\bq = -k \grad u,$$
where $u$ is the temperature, $\bq$ is the heat flux, and $k>0$ is the conductivity
\item consider a solid with a heat capacity $c>0$ (heat energy gained by increase in temperature) and a mass density $\rho>0$ (mass per unit volume)
\item given a heat source $f$ within the domain $\Omega$, conservation of energy says
    $$c\rho\, \frac{\partial u}{\partial t} = - \Div\bq + f \quad \iff \quad \frac{d}{dt}\left(\int_\Omega c\rho\, u \,dx\right) = - \int_{\partial\Omega} \bq\cdot\bn\,ds + \int_\Omega f \,dx$$
\item assume steady state, so $\partial/\partial t=d/d t=0$
\item if $k$ is constant then $- \Div\bq = - \Div(-k \grad u) = k \grad^2 u$
\item we derive Poisson's equation, the (steady) \emph{heat} or \emph{diffusion equation}
    $$0 = k \grad^2 u + f \qquad \text{ within } \Omega$$
\item boundary conditions are needed!  (coming soon)
\end{itemize}
\end{block}
\end{frame}


\begin{frame}{applications}

\begin{block}{application: deflection of an elastic membrane}
\begin{itemize}
\small
\item suppose $z=u(x,y)$ is a vertical displacement of an elastic membrane over a domain $\Omega\subset \RR^2$
\item suppose that along $\partial \Omega$ we have put a rigid frame with height $z=g(x,y)$
\item suppose that the slopes along the graph of $g$ are small
\item suppose that the membrane is loaded by a force $f$ (per unit area)
\item group the elastic properties of the membrane into a constant $\kappa>0$
\item then the vertical displacement satisfies Poisson's equation
    $$- \kappa \grad^2 u = f \qquad \text{ within } \Omega$$
subject to $u=g$ along $\partial \Omega$
\item the standard derivation starts from a the total elastic energy of the membrane, in which case the Poisson equation is the condition that the derivative is zero at the minimizer
\end{itemize}
\end{block}
\end{frame}

\section{the Poisson equation in weak form: $\ds \int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx$}

\begin{frame}{deriving the weak form of the Poisson equation}

\begin{itemize}
\item first I'll describe \emph{how} to get the weak formulation, then I'll explain \emph{why}
\item let's start from the Poisson equation, $-\grad^2 u=f$
\item we seek the solution function $u:\Omega\to\RR$
\item remember that $\grad^2=\Div\grad$, so the problem is:
    $$- \Div(\grad u) = f \qquad \text{ within } \Omega$$
\item suppose $v:\Omega\to\RR$ is another function, which we call a \emph{test function}
\item multiply the Poisson equation by $v$ and integrate over $\Omega$:
    $$- \int_\Omega \Div(\grad u) v\,dx \stackrel{\ast}{=} \int_\Omega f v\,dx$$
\item apply the integration by parts rule from the \href{https://bueler.github.io/fa/assets/slides/S26/week10.pdf}{weeks 10\&11 slides}:
    $$\int_\Omega \phi\,\Div \bV\,dm = \int_{\partial\Omega} \phi\,\bV\cdot \bn\,ds - \int_\Omega \grad \phi\cdot \bV\,dm$$

   \begin{itemize}
   \item[$\circ$] we'll apply this on the left side of $\ast$, with $\phi=v$ and $\bV=\grad u$
   \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{deriving the weak form of the Poisson equation}

\begin{itemize}
\item we get:
    $$-\int_{\partial\Omega} v\,(\grad u)\cdot \bn\,ds + \int_\Omega \grad v\cdot \grad u\,dx = \int_\Omega f v\,dx$$
\item rearrange to get:
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx + \alert{\int_{\partial\Omega} v\,(\grad u\cdot \bn)\,ds}$$

   \begin{itemize}
   \item[$\circ$] the integral $\int_\Omega f v\,dx$ is \emph{data}, because $f$ is given
   \item[$\circ$] existence and uniqueness of the solution $u$, essentially requires turning the \alert{boundary integral} into data also
   \end{itemize}
\item there are two options for removing $u$ from the boundary integral:

\begin{enumerate}
\item for Dirichlet boundary conditions, only use test functions $v$ which have zero value along $\partial \Omega$
\item for Neumann boundary conditions, replace $\grad u\cdot \bn$ with the data $g_N$
\end{enumerate}
\end{itemize}
\end{frame}


\begin{frame}{\emph{ALMOST} the weak form of the Poisson equation}

\begin{block}{weak formulation of the Poisson equation}
Let $\Omega\subset \RR^d$ be an open, bounded domain with Lipschitz boundary, and suppose $f\in L^2(\Omega)$ is given.  Solve one of these problems:
\begin{enumerate}
\item Find a solution $u\in C_c^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \hspace{20mm} \text{(homogeneous Dirichlet)}$$
for all test functions $v\in C_c^1(\Omega)$.
\item Find a solution $u\in C^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx + \int_{\partial\Omega} g_N v\,ds \hspace{10mm} \text{(general Neumann)}$$
for all test functions $v\in C^1(\Omega)$.
\end{enumerate}
\end{block}

\begin{itemize}
\item \alert{this is not the right way to do it!}
\item asking for $C^1(\Omega)$ is too strong!
\end{itemize}
\end{frame}


\begin{frame}{the weak form of the Poisson equation}

\begin{block}{weak formulation of the Poisson equation}
Let $\Omega\subset \RR^d$ be an open, bounded domain with Lipschitz boundary, and suppose $f\in L^2(\Omega)$ is given.  Solve one of these problems:
\begin{enumerate}
\item Find a solution \alert{$u\in H_0^1(\Omega)$} so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \hspace{20mm} \text{(homogeneous Dirichlet)}$$
for all test functions \alert{$v\in H_0^1(\Omega)$}.
\item Find a solution \alert{$u\in H^1(\Omega)$} so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx + \int_{\partial\Omega} g_N v\,ds \hspace{10mm} \text{(general Neumann)}$$
for all test functions \alert{$v\in H^1(\Omega)$}.
\end{enumerate}
\end{block}

\begin{itemize}
\item \alert{I will define the \emph{Sobolev space} $H^1(\Omega)$ in the next section}
\item for general Dirichlet conditions there is an additional complication
\end{itemize}
\end{frame}


\begin{frame}{what is the ``weak formulation''?}

\begin{block}{observation 1.}
The strong form is pointwise: $(-\grad^2 u)(x) = f(x)$ holds at every $x\in \Omega$.  By contrast, \alert{the weak form is about the average using a test function}.  That is, the weak form holds on the support of $v$.
\end{block}

\begin{itemize}
\item here's a cartoon of where the equation is enforced:
\end{itemize}

\vspace{25mm}

\begin{block}{observation 2.}
The strong form refers to the second derivatives, the Laplacian, of the unknown solution $u$.  By contrast, \alert{only the first derivatives of $u$}, namely the gradient, \alert{is needed in the weak form}.
\end{block}
\end{frame}


\begin{frame}{general Dirichlet boundary conditions}

\begin{itemize}
\item Suppose $\Omega\subset \RR^d$ is an open, bounded domain with Lipschitz boundary.
\item Suppose $f\in L^2(\Omega)$ and $g_D\in L^2(\partial\Omega)$ are given.
\item Recall the strong form of the general Dirichlet problem:
   $$- \grad^2 u = f \quad \text{ on } \Omega, \quad \text{ and }
u = g_D \quad \text{ on } \partial \Omega$$
\item To build the general Dirichlet weak form we substitute $u+u_D$ into the homogeneous weak form.
\end{itemize}

\begin{block}{weak form general Dirichlet problem}
Let $u_D \in H^1(\Omega) \cap C(\bar\Omega)$ be any function, \emph{defined on the whole domain} $\Omega$, so that $u_D=g_D$ along $\partial\Omega$.  Find a solution $u\in H_0^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx - \int_\Omega \grad u_D \cdot \grad v\,dx$$
for all test functions $v\in H_0^1(\Omega)$.
\end{block}

\begin{itemize}
\item The new right-hand side is data.
\end{itemize}
\end{frame}

\item suppose $\Omega \subset \RR^2$ is a polygon
\item put a mesh $\mathcal{T}$ of triangles on $\Omega$
\item denote nodes $x_i$ and triangles $T_k$

\item we call $f(x,y)=a + bx + cy$ a \emph{linear polynomial} in two variables

\begin{definition}[the continuous $P_1$ finite element space over $\mathcal{T}$]
    $$P_1 = \left\{f \in C(\Omega)\,:\,f\big|_{T_k} \text{ is a linear polynomial}\right\}$$

\begin{frame}{what is a finite element method (FEM)?}

\begin{block}{weak form of the Poisson equation (homogeneous Dirichlet)}
Find a solution $u\in V = H_0^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \qquad \forall v\in V \hspace{15mm}$$
\end{block}

\begin{itemize}
\item FEM solves the \alert{same weak form}, but over the finite-dim'l subspace $V_h$
\end{itemize}

\begin{block}{a FEM for the Poisson equation (homogeneous Dirichlet)}
Find a solution $u_h\in V_h =\left\{f \in P_1\,:\,f\big|_{\partial\Omega} = 0\right\}$ so that
    $$\int_\Omega \grad u_h \cdot \grad v_h\,dx = \int_\Omega f v_h\,dx \qquad \forall v_h\in V_h \hspace{20mm}$$
\end{block}

\section{defining the Sobolev spaces: $H^k(\Omega)$}

\begin{frame}{recall weak derivatives}

\begin{itemize}
\item given a bounded open set $\Omega\subset \RR^d$
\item recall the definition: $L^2(\Omega) = \{f:\Omega\to\RR\,\big|\,\int_\Omega |f|^2 <\infty\}$
   \begin{itemize}
   \item[$\circ$] if $\Omega$ is bounded then $L^2(\Omega) \subset L^1(\Omega)$
   \end{itemize}
\item recall our multi-index derivative notation $D^\alpha \varphi$, from the \href{https://bueler.github.io/fa/assets/slides/S26/week10.pdf}{weeks 10\&11 slides}, and the main definition:
\end{itemize}

\begin{definition}[weak derivative]
\small
if $f\in \Lloc(\Omega)$, and if $g\in \Lloc(\Omega)$ satisfies
    $$\int_\Omega g\, \varphi\,dm = (-1)^{|\alpha|} \int_\Omega f\, D^\alpha\varphi\,dm \qquad \text{for all } \varphi \in C_c^\infty(\Omega)$$
then we say $g$ is the \emph{$\alpha$th weak derivative} of $f$, and we write $D^\alpha f = g$ a.e.
\end{definition}

\begin{itemize}
\item \emph{example:} $f$ has a weak derivative $g=\partial f/\partial x_k \in L^1(\Omega)$ if
    $$\int_\Omega g\, \varphi\,dm = - \int_\Omega f\, \frac{\partial\varphi}{\partial x_k}\,dm \qquad \text{for all } \varphi \in C_c^\infty(\Omega)$$
and $\int_\Omega|g|\,dm < \infty$
\end{itemize}
\end{frame}


\begin{frame}{first Sobolev space}

\begin{definition}[Sobolev space $H^1(\Omega)$]
    $$H^1(\Omega) = \left\{f\in L^2(\Omega) \,\Big|\, \begin{matrix}  \text{the weak partial derivatives } \frac{\partial f}{\partial x_k} \text{ exist} \\ \text{and are in } L^2(\Omega) \text{ for } k=1,\dots,d \end{matrix} \right\}$$
is a Hilbert space with inner product:
	$$\ip{f}{g}_{H^1} = \int_\Omega f(x)g(x)\,dx + \int_\Omega \grad f(x)\cdot \grad g(x)\,dx$$
and norm: \quad $\|f\|_{H^1} = \sqrt{\ip{f}{f}_{H^1}} = \left(\int_\Omega |f|^2 + \int_\Omega |\grad f|^2\right)^{1/2}$
\end{definition}

\begin{itemize}
\small
\item \emph{example on $\RR^2$:} Consider the unit square $\Omega = (0,1)^2$ and the function $f(x)=x_1 + x_2$.  It has gradient $\grad f = \ip{1}{1}$, and norm $\|f\|_{H^1} = \sqrt{19/6}$:
\begin{align*}
\|f\|_{H^1}^2 &= \int_\Omega |x_1 + x_2|^2\,dx + \int_\Omega |\ip{1}{1}|^2\,dx \\
&= \int_0^1 \int_0^1 x_1^2 + 2 x_1 x_2 + x_2^2 \,dx_1\,dx_2 + \int_0^1 \int_0^1 2 \,dx_1\,dx_2 = \frac{7}{6} + 2 = \frac{19}{6}
\end{align*}
\end{itemize}
\end{frame}


\begin{frame}{more Sobolev spaces}

\begin{definition}[Sobolev space $H^k(\Omega)$]
    $$H^k(\Omega) = \left\{f\in L^2(\Omega) \,\Big|\, \begin{matrix}  \text{the weak derivatives } D^\alpha f \text{ exist} \\ \text{and are in } L^2(\Omega) \text{ for all } |\alpha|\le k\end{matrix} \right\}$$
is a Hilbert space with inner product:
	$$\ip{f}{g}_{H^k} = \sum_{|\alpha|\le k} \int_\Omega (D^\alpha f)(x) (D^\alpha g)(x)\,dx$$
and norm: \quad $\|f\|_{H^k} = \sqrt{\ip{f}{f}_{H^k}}$
\end{definition}

\begin{itemize}
\item ``$H$\,'' is for Hilbert
\item we may write $H^0(\Omega) = L^2(\Omega)$, because $D^\alpha f = f$ if $\alpha=(0,\dots,0)$
\item some sources use notation $H^k(\Omega) = W^{k,2}(\Omega)$
   \begin{itemize}
   \item[$\circ$] the ``$2$'' indicates that the weak derivatives are in $L^2(\Omega)$
   \item[$\circ$] general case: $W^{k,p}(\Omega)$ has weak derivatives in $L^p(\Omega)$ up to order $k$
   \end{itemize}
\end{itemize}
\end{frame}


\section{more about $H^1(\Omega)$}

\begin{frame}{$H^1$ on an interval}

\begin{itemize}
\item if $\Omega=(a,b)$ is an interval of the real line, then we have \alert{a simple definition using ordinary derivatives}
\end{itemize}

\begin{definition}
    $$H^1(a,b) = \left\{f\in L^2(a,b) \,\Big|\, \begin{matrix}  \text{the weak derivative } f' \text{ exists} \\ \text{and } f' \in L^2(a,b) \end{matrix} \right\}$$
\end{definition}

\begin{itemize}
\item the norm is: \quad $\ds \left(\|f\|_{H^1}\right)^2 = \int_a^b |f(x)|^2 \, dx + \int_a^b |f'(x)|^2\,dx$
\item but recall:
    \begin{itemize}
    \item[$\circ$] $L^2(a,b) \subset L^1(a,b)$, since $m((a,b))<\infty$, and
    \item[$\circ$] $\ds f' \in L^1(a,b) \quad \implies \quad f \in AC(a,b) \quad \text{and}  \quad f(x) = f(a) + \int_a^x f'(t)\,dt$
    \end{itemize}
\item most importantly:
	$$\alert{H^1(a,b) \subset C^0([a,b])}$$

    \begin{itemize}
    \item[$\circ$] and $C^1([a,b]) \subsetneq H^1(a,b) \subsetneq C^0([a,b]) \subsetneq L^2(a,b)$
    \end{itemize}
\end{itemize}

\begin{frame}{$H^1$ on an interval, via Fourier series}

\begin{block}{{\small recall Fourier series in $L^2(-\pi,\pi)$}}
{\footnotesize
using ON sequence $\frac{1}{\sqrt{2\pi}} e^{inx}$, $n\in\ZZ$, we have convergence in the $\|\cdot\|_2$ norm:
	$$f(x) = \frac{1}{\sqrt{2\pi}}\sum_{n=-\infty}^\infty c_n e^{inx} \quad \text{with} \quad c_n = \frac{1}{\sqrt{2\pi}} \int_{-\pi}^\pi f(x) e^{-inx}\,dx,$$
Parseval's equality: \quad $\ds \|f\|_{L^2}^2 = \sum_{n=-\infty}^\infty |c_n|^2$
}
\end{block}

\begin{itemize}
{\small
\item but now consider the derivative formula: \quad $\ds f'(x) = \frac{1}{\sqrt{2\pi}} \sum_{n=-\infty}^\infty i n c_n e^{inx}$
\item \dots and we can norm the derivative (\emph{use orthonormality}):
{\footnotesize
\begin{align*}
\int_{-\pi}^\pi |f'(x)|^2\,dx &= \sum_{k,n} \frac{1}{2\pi} \int_{-\pi}^\pi (i n c_n\, e^{inx})(\overline{i k c_k\, e^{ikx}})\,dx = \sum_{k,n} n k c_n c_k \delta_{kn} = \sum_{n=-\infty}^\infty n^2 |c_n|^2
\end{align*}
}
}
\end{itemize}

{\small
\begin{lemma}[$H^1(-\pi,\pi)$ via Fourier series]
    $$H^1(-\pi,\pi) = \left\{f\in L^2(-\pi,\pi) \,\Big|\, \sum_{n=-\infty}^\infty n^2 |c_n|^2 < \infty\right\}$$
\end{lemma}
}
\end{frame}


\begin{frame}{$H^k$ on an interval, via Fourier series}

\begin{itemize}
\item the same logic applies for higher derivatives, e.g.:
{\small
\begin{align*}
\int_{-\pi}^\pi |f''|^2 &= \sum_{k,n} \frac{1}{2\pi} \int_{-\pi}^\pi ((i n)^2 c_n\, e^{inx})(\overline{(i k)^2 c_k\, e^{ikx}})\,dx = \sum_{n\in\ZZ} n^4 |c_n|^2
\end{align*}
}

\vspace{-2mm}
\begin{lemma}[$H^k(-\pi,\pi)$ via Fourier series]
    $$H^k(-\pi,\pi) = \Big\{f\in L^2(-\pi,\pi) \,\Big|\, \sum_{n\in\ZZ} n^{2k} |c_n|^2 < \infty\Big\}$$
\end{lemma}

\item with norm formula, by Parseval:
{\small
\begin{align*}
\|f\|_{H^k}^2 &= \int_{-\pi}^\pi |f|^2 + \dots + \int_{-\pi}^\pi |f^{(k)}|^2 \\
&= \sum_{n\in\ZZ} (1+ n^2 + \dots + n^{2k}) |c_n|^2 = \sum_{n\in\ZZ} \left(\sum_{j=0}^k n^{2j}\right) |c_n|^2
\end{align*}
}
\item we could make the lemma a definition, and \alert{not need weak derivatives}
\item also: \alert{$k$ does not need to be an integer}, but we will not pursue this
\end{itemize}
\end{frame}


\begin{frame}{are functions in $H^1(\Omega)$ continuous for $\Omega\subset\RR^2$?}

is the unit disc, and $r=|x|$
\item the first two examples are expected

\begin{block}{a continuous function in $H^1(D)$ \dots \emph{expected}}

\end{align*}
}
\end{block}

\begin{frame}{\emph{fact:} \quad $\|\grad f\|_{L^2}$ is equivalent to $\|f\|_{H^1}$ in $H_0^1(\Omega)$}

\begin{itemize}
\item once you have zero boundary values, then \alert{the only way for the function to get large is for the gradient to also be large}

\begin{theorem}[Poincar\'e-Friedrichs inequality]
Let $\Omega\subset \RR^d$ be an open, bounded domain with Lipschitz boundary, or an open set which is bounded in one direction.  There is $C>0$, depending on $\Omega$, so that
	$$\|f\|_{L^2} \le C \|\grad f\|_{L^2} \quad \text{for all } f\in H_0^1(\Omega).$$
\end{theorem}

\item main consequence: the simpler formula
	$$|f|_{H^1} = \|\grad f\|_{L^2} = \left(\int_\Omega |\grad f|^2\right)^{1/2}$$
is equivalent to the full norm $\|f\|_{H^1} = \left(\int_\Omega |f|^2 + \int_\Omega |\grad f|^2\right)^{1/2}$ on $H_0^1(\Omega)$
    \begin{itemize}
    \item[$\circ$] it is \emph{not} equivalent on the whole space $H^1(\Omega)$
    \end{itemize}
\item $|f|_{H^1}$ is called the \emph{$H^1$ seminorm}
    \begin{itemize}
    \item[$\circ$] with the understanding that it is actually a norm on $H_0^1(\Omega)$
    \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{Poisson equation Dirichlet problems: weak forms}

\begin{itemize}
\item now that $H_0^1(\Omega)$ is defined, we can make sense of these problems
\end{itemize}

\begin{block}{weak formulation of Dirichlet problems}
Let $\Omega\subset \RR^d$ be an open, bounded domain with Lipschitz boundary, and suppose $f\in L^2(\Omega)$ is given.  Solve:
\begin{enumerate}
\item \alert{homogeneous Dirichlet}: Find $u\in H_0^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx$$
for all test functions $v\in H_0^1(\Omega)$.
\item \alert{general Dirichlet}: Let $u_D \in H^1(\Omega) \cap C(\bar\Omega)$ be any function so that $u_D=g_D$ along $\partial\Omega$.  Find $\tilde u\in H_0^1(\Omega)$ so that
    $$\int_\Omega \grad \tilde u \cdot \grad v\,dx = \int_\Omega f v\,dx - \int_\Omega \grad u_D \cdot \grad v\,dx$$
for all test functions $v\in H_0^1(\Omega)$.  The solution is $u=\tilde u + u_D \in H^1(\Omega)$.
\end{enumerate}
\end{block}
\end{frame}
