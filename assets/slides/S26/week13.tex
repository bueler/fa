\section{recall: strong and weak forms of the Poisson equation}


\begin{frame}{recall: boundary-value problems for the Poisson equation}

\begin{itemize}
\item suppose $\Omega\subset \RR^d$ is open and has a Lipschitz, or continuously- differentiable, boundary $\partial\Omega$, with outward unit normal vector field $\bn$
\end{itemize}

\begin{block}{general Dirichlet problem (\emph{strong form})}
for $f\in L^2(\Omega)$ and $g_D\in C^1(\partial\Omega)$, find $u:\Omega\to \RR$ so that
\begin{align*}
- \grad^2 u &= f \quad \text{ on } \Omega \\
u &= g_D \quad \text{ on } \partial \Omega
\end{align*}
\end{block}

\begin{block}{general Neumann problem (\emph{strong form})}
for $f\in L^2(\Omega)$ and $g_N\in L^2(\partial\Omega)$, find $u:\Omega\to \RR$ so that
\begin{align*}
- \grad^2 u &= f \quad \text{ on } \Omega \\
\grad u\cdot \bn &= g_N \quad \text{ on } \partial \Omega
\end{align*}
\end{block}

\begin{itemize}
\item other notation: $\grad u\cdot \bn = \frac{\partial u}{\partial n} = u_n$
\end{itemize}
\end{frame}


\begin{frame}{recall: the Sobolev spaces $H^1(\Omega)$ and $H_0^1(\Omega)$}

\small

\begin{definition}
    $$H^1(\Omega) = \left\{f\in L^2(\Omega) \,\Big|\, \begin{matrix} \text{the weak derivative } f' \text{ exists} \\ \text{and is in } L^2(\Omega)\end{matrix}\right\}$$
is a Hilbert space with inner product and norm:
	$$\ip{f}{g}_{H^1} = \int_\Omega f(x) g(x)\,dx + \int_\Omega f'(x) g'(x)\,dx, \qquad \|f\|_{H^1} = \sqrt{\ip{f}{f}_{H^1}}$$
\end{definition}

\begin{definition}
    $$H_0^1(\Omega) = \overline{C_c^1(\Omega)},$$
where closure is in $H^1$ norm, is a closed subspace of $H^1(\Omega)$
\end{definition}
\end{frame}


\begin{frame}{recall: the weak forms of the Poisson problems}

\begin{block}{weak formulation of the Poisson equation}
Suppose $f\in L^2(\Omega)$ and $g_N\in L^2(\partial\Omega)$ are given.  Solve one of these problems:
\begin{enumerate}
\item \alert{homogeneous Dirichlet:}  Find a solution $u\in H_0^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \quad \text{ for all test functions } v\in H_0^1(\Omega)$$
\item \alert{general Neumann:}  Find a solution $u\in H^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx + \int_{\partial\Omega} g_N v\,ds  \quad \text{ for all test functions } v\in H^1(\Omega)$$
\end{enumerate}
\end{block}
\end{frame}


\begin{frame}{recall: general Dirichlet boundary conditions}

\begin{itemize}
\item general Dirichlet conditions have an additional complication
\end{itemize}

\begin{block}{weak form: \alert{general Dirichlet}}
Suppose $f\in L^2(\Omega)$ and $g_D\in C^1(\partial\Omega)$ are given.  Let $u_D \in H^1(\Omega) \cap C(\bar\Omega)$ be any function, defined on the whole domain $\Omega$, so that $u_D=g_D$ along $\partial\Omega$.  Find a solution $u\in H_0^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx - \int_\Omega \grad u_D \cdot \grad v\,dx$$
for all test functions $v\in H_0^1(\Omega)$.
\end{block}

\begin{itemize}
\item the actually solution to the problem is $u+u_D$
\item even more generally, ``mixed'' conditions could partition the boundary: $\partial \Omega = \partial_D \Omega \cup \partial_N \Omega$
\end{itemize}

\end{frame}

\begin{frame}{real Fourier series}

\small

\begin{lemma}  The Fourier series $g(x) = \frac{1}{\sqrt{2\pi}} \sum_{n\in\ZZ} c_n e^{inx}$ is a real-valued function if and only if $c_0\in\RR$ and $c_{-n} = \overline{c_n}$.
\end{lemma}

\begin{corollary}  If $g(x)$ is a real-valued function, and if for $n\ge 0$ we write $c_n = \frac{\sqrt{2\pi}}{2} (a_n-i b_n)$, with $a_n,b_n\in\RR$, then \, $\ds g(x) = \frac{a_0}{2} + \sum_{n=1}^\infty a_n \cos(n\theta) + b_n \sin(n\theta)$.
\end{corollary}

\footnotesize
\noindent \emph{Proof.} Write $g(x) = \frac{1}{\sqrt{2\pi}} c_0 + \frac{1}{\sqrt{2\pi}} \sum_{n=1}^\infty \left(c_n e^{inx} + c_{-n} e^{-inx}\right)$.  Then use the lemma, giving
	$$g(x) = \frac{1}{\sqrt{2\pi}} c_0 + \frac{1}{\sqrt{2\pi}} \sum_{n=1}^\infty 2 \Re \left(c_n e^{inx}\right)$$
Substitute $c_n = \frac{\sqrt{2\pi}}{2} (a_n-i b_n)$ for $n\ge 0$, and simplify, to get the result.\qed
\small

\begin{lemma}
    $$a_n = \frac{1}{\pi} \int_{-\pi}^\pi g(x)\cos(nx)\,dx, \qquad b_n = \frac{1}{\pi} \int_{-\pi}^\pi g(x)\sin(nx)\,dx$$
\end{lemma}
\end{frame}


\begin{frame}{harmonic functions on the disk}

\begin{itemize}
\item on the other hand, recall that the real and imaginary parts of any complex-analytic function on the disk are harmonic
\item for example, the monomials $z^n$, $n\ge 0$, are analytic
\item also, $z=r e^{i\theta}$ in polar coordinates, so the real and imaginary parts of
    $$z^n=r^n e^{in\theta}=r^n(\cos(n\theta) + i \sin(n\theta))$$
are harmonic
\end{itemize}

\begin{definition} for $n\ge 0$, define these \alert{harmonic functions} on $D$:
$$\phi_n(r,\theta) = r^n \cos(n\theta), \quad \psi_n(r,\theta) = r^n \sin(n\theta)$$
\end{definition}

\footnotesize
\noindent \emph{Proof they are harmonic.}  In polar coordinates, $\alert{\grad^2 u = u_{rr} + \frac{1}{r} u_r + \frac{1}{r^2} u_{\theta\theta}}$.  Then for $u=\phi_n$,
\begin{align*}
\grad^2 u &= n (n-1) r^{n-2} \cos(n\theta) + \frac{1}{r} n r^{n-1} \cos(n\theta) + \frac{1}{r^2} (-n^2 r^n \cos(n\theta)) \\
  &= \Big[n (n-1) + n - n^2\Big] r^{n-2} \cos(n\theta) = 0 r^{n-2} \cos(n\theta) = 0
\end{align*}
Similarly for $u=\psi_n$. \hfill \qed
\end{frame}


\begin{frame}{Laplace's equation (Dirichlet's problem) on a disk}

\begin{itemize}
\item combine these ideas to solve Laplace's equation $\grad^2 u = 0$ on the disk
\item start with Fourier series for $g\in L^2$ along the boundary; for $x\in (-\pi,\pi]$:
    $$g(x) = \frac{a_0}{2} + \sum_{n=1}^\infty a_n \cos(n\theta) + b_n \sin(n\theta)$$
\item use the coefficients to define a function on $\bar D$; for $0\le r \le 1$, $\theta \in (-\pi,\pi]$:
\begin{align*}
u(r,\theta) &= \frac{a_0}{2} + \sum_{n=1}^\infty a_n r^n \cos(n\theta) + b_n r^n \sin(n\theta) \quad \text{ for } (r,\theta) \in \bar D
\end{align*}
\end{itemize}

\vspace{-3mm}
\begin{theorem}  Let $g\in L^2(\partial D)$.  Then $u$ is continuous on $D$ and satisfies $\grad^2 u=0$ within $D$.  And $\ds \lim_{r\to 1^-} u(r,\theta) = g(\theta)$ with convergence in $L^2$.
\end{theorem}

\footnotesize
\noindent \emph{Proof.}  The series converges uniformly on every compact set $K\subset D$, the terms are continuous, so $u$ is continuous on $D$.  For $r<1$, compute $\|u(r,\cdot) - g(\cdot)\|_{L^2}$.  Show it goes to 0 as $r\to 1$ by comparing a geometric series. \hfill \qed
\end{frame}

\begin{frame}{Poisson on a disk}

\begin{itemize}
\item now start from series expansion of source function:
$$f(r,\theta) = \sum_{n=1}^\infty a_{0n}\, \phi_{0n}(r,\theta) + \sum_{m,n=1}^\infty \big(a_{mn}\, \phi_{mn}(r,\theta) + b_{mn}\, \psi_{mn}(r,\theta)\big)$$
\item solve $-\grad^2 u = f$ using the eigenfunction property $\grad^2 \phi_{mn} = - \lambda_{mn}^2 \phi_{mn}$

\begin{theorem}  Suppose $f\in L^2(D)$ is expanded in $\phi_{mn}$, $\psi_{mn}$ as on the last slide, with convergence in $L^2$.  Define
\begin{equation*}
u(r,\theta) = \sum_{n=1}^\infty \frac{a_{0n}}{\lambda_{0n}^2}\, \phi_{0n}(r,\theta) + \sum_{m,n=1}^\infty \frac{1}{\lambda_{mn}^2} \big(a_{mn}\, \phi_{mn}(r,\theta) + b_{mn}\, \psi_{mn}(r,\theta)\big).
\end{equation*}
Then $u\in H_0^1(D) \cap C^2(D)$ solves the both the strong form and the weak form of the Poisson equation
    $$-\grad^2 u = f$$
\end{theorem}

\begin{frame}{spectral expansions for Poisson equation}

\begin{itemize}
\item we can combine the above techniques to solve the general Dirichlet problem for the Poisson equation on the disk:

   \item[$\circ$] method: (i) solve $-\grad^2 u_1 = 0$ and $u_1|_{\partial D}=g$ from Fourier series expansion of $g$, (ii) solve $-\grad^2 u_2 = f$ and $u_2|_{\partial D}=0$ from eigenfunction expansion of $f$, and (iii) then $u=u_1+u_2$

\item \dots only possible when the domain $\Omega$ has lots of symmetry, like the disk

   \item[$\circ$] also called \alert{separation of variables} (for PDEs)
   \item[$\circ$] essentially the domain has to be \emph{rectangular} in the preferred coordinates
   \item[$\circ$] for example, $D \cong [0,1) \times (-\pi,\pi]$ in the $r,\theta$ variables
   \end{itemize}

\section{well-posedness for Poisson equation problems}

\begin{frame}{well-posedness of math problems}

\begin{definition}
a mathematical problem is \alert{well-posed} if the solution satisfies the following properties:
\begin{enumerate}
\item it exists
\item it is unique
\item its value changes continuously as the data of the problem changes
\end{enumerate}
\end{definition}

\small
\begin{block}<2>{Example: finite-dimensional linear systems}
Suppose $A\in \CC^{m\times m}$ is invertible and $\bb\in\CC^m$.  The problem of solving \only<1>{$A\bx = \bb$}\only<2>{$\boxed{A\bx = \bb}$} for

$\bx\in\CC^m$ is well posed because
\begin{enumerate}
\item exists: $\bx = A^{-1}\bb$
\item unique: if $A\bx_1=\bb$ and $A\bx_2=\bb$ then $A(\bx_1-\bx_2)=\bzero$ so $\bx_1=\bx_2$
\item continuous in the data $\bb$: if $A\bx_1=\bb_1$ and $A\bx_2=\bb_2$ then

\quad $\|\bx_1-\bx_2\| = \|A^{-1}\bb_1 - A^{-1}\bb_2\| \le \|A^{-1}\| \|\bb_1 - \bb_2\|$
\end{enumerate}
\end{block}
\end{frame}


\begin{frame}{are Poisson problems well-posed?}

\begin{itemize}
\item so, how do we prove that the Poisson equation is a well-posed problem?
\item solving it in particular cases, like we have for the disk, will not work
\item we already have two important theorems which will help
\end{itemize}

\begin{theorem}[Poincar\'e-Friedrichs inequality]
Let $\Omega\subset \RR^d$ be an open, bounded domain with Lipschitz boundary, or an open set which is bounded in one direction.  There is $C>0$, depending on $\Omega$, so that
	$$\|f\|_{L^2} \le C \|\grad f\|_{L^2} \quad \text{for all } f\in H_0^1(\Omega).$$
\end{theorem}

\begin{theorem}[Riesz-Fr\'echet representation theorem]
If $(V,\ip{\cdot}{\cdot})$ is a (real) Hilbert space, and if $\ell \in V'$ is a bounded linear functional, then there exists $u\in V$ so that
	$$\ip{u}{v} = \ell(v) \quad \text{for all } v\in V$$
\end{theorem}
\end{frame}


\begin{frame}{Riesz representation route to well-posedness 1}

\begin{theorem}
The weak-form homogeneous Dirichlet problem, for $u\in H_0^1(\Omega)$ such that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \quad \text{ for all } v\in H_0^1(\Omega),$$
is well-posed.
\end{theorem}

\small
\emph{Proof.} Let $\iip{u}{v} = \int_\Omega \grad u \cdot \grad v\,dx$.  We claim this is also an inner product on $H_0^1(\Omega)$.\footnote{The usual inner product is \, $\ip{u}{v} = \int_\Omega u v\,dx + \int_\Omega \grad u \cdot \grad v\,dx$.}  It is clearly nonnegative, symmetric, and bilinear.  It is nondegenerate because $\iip{u}{u}=0$ implies
	$$\|u\|_{H^1}^2 = \int_\Omega u^2\,dx + \int_\Omega |\grad u|^2\,dx \le (C^2+1)\int_\Omega |\grad u|^2\,dx = (C^2+1) \iip{u}{u} = 0,$$
by the Poincar\'e-Friedrichs inequality, so $u=0$.  Thus $(H_0^1(\Omega),\iip{\cdot}{\cdot})$ is a real Hilbert space.
\end{frame}


\begin{frame}{Riesz representation route to well-posedness 2}

\small
\emph{Proof, continued.} 
On the other hand, the right-hand side defines a linear functional, $\ell:H_0^1(\Omega)\to \RR$, with formula $\ell(v)=\int_\Omega fv\,dx$.  This is bounded by Cauchy-Schwarz:
    $$|\ell(v)| \le \int_\Omega |fv|\,dx \le \|f\|_{L^2}\|v\|_{L^2} \le \|f\|_{L^2}\|v\|_{H^1}.$$

By the Riesz-Fr\'echet representation theorem there is a unique $u\in H_0^1(\Omega)$ so that
   $$\iip{u}{v} = \ell(v),$$
which says exactly that the weak form problem has a unique solution.

Finally, to show that the solution is continuous with respect to the data $f$, observe that the solution process is a well-defined linear map.  That is, there is a linear map $T:L^2(\Omega) \to H_0^1(\Omega)$ defined by\footnote{Conceptually, $T=(-\grad^2)^{-1}$.  But boundary conditions are involved!}
	$$T f = u$$
Recall that a linear map is continuous if and only if it is bounded.  We will show that
    $$\|u\|_{H^1} = \|T f\|_{H^1} \le c \|f\|_{L^2}.$$
\end{frame}


\begin{frame}{Riesz representation route to well-posedness 3}

\small
\emph{Proof, continued.}
To show that $T$ is bounded we substitute $u$ for $v$ in the weak form, use a previous calculation, and apply Cauchy-Schwarz:
\begin{align*}
\|u\|_{H^1}^2 &\le (C^2+1) \iip{u}{u} = (C^2+1) \int_\Omega \grad u \cdot \grad u\,dx \\
  &= (C^2+1) \int_\Omega f u\,dx \le (C^2+1) \|f\|_{L^2} \|u\|_{L^2} \le (C^2+1) \|f\|_{L^2} \|u\|_{H^1}
\end{align*}
Now divide by the norm of $u$ to get the desired bound:
    $$\|u\|_{H^1} \le (C^2+1) \|f\|_{L^2}.$$
This shows that the solution map $Tf=u$ is continuous, and concludes the proof of well-posedness. \qed
\end{frame}


\begin{frame}{the Neumann problem is \emph{not} well-posed}

\begin{itemize}
\item recall the general Neumann problem is to find $u\in H^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx + \int_{\partial\Omega} g_N v\,ds  \quad \text{ for all } v\in H^1(\Omega)$$

    \begin{itemize}
    \item[$\circ$] strong form:\,\, $- \grad^2 u = f$ on $\Omega$,\, $\grad u\cdot \bn = g_N$ on $\partial \Omega$
    \end{itemize}
\item this has a uniqueness problem!
\end{itemize}

\begin{lemma}
If the weak-form general Neumann problem has a solution then it has infinitely-many solutions.
\end{lemma}

\small
\emph{Proof.}  Let $u\in H^1(\Omega)$ be a solution and consider any $C\in\RR$.  Then $\tilde u(x)=u(x)+C$ is also in $H^1(\Omega)$, and it satisfies the same weak form. \hfill \qed
\end{frame}
