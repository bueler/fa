\section{recall: strong and weak forms of the Poisson equation}


\begin{frame}{recall: boundary-value problems for the Poisson equation}

\begin{itemize}
\item suppose $\Omega\subset \RR^d$ is open and has a Lipschitz, or continuously- differentiable, boundary $\partial\Omega$, with outward unit normal vector field $\bn$
\end{itemize}

\begin{block}{general Dirichlet problem (\emph{strong form})}
for $f\in L^2(\Omega)$ and $g_D\in C^1(\partial\Omega)$, find $u:\Omega\to \RR$ so that
\begin{align*}
- \grad^2 u &= f \quad \text{ on } \Omega \\
u &= g_D \quad \text{ on } \partial \Omega
\end{align*}
\end{block}

\begin{itemize}
\item called \emph{homogeneous Dirichlet} if $g_D=0$
\end{itemize}

\begin{block}{general Neumann problem (\emph{strong form})}
for $f\in L^2(\Omega)$ and $g_N\in L^2(\partial\Omega)$, find $u:\Omega\to \RR$ so that
\begin{align*}
- \grad^2 u &= f \quad \text{ on } \Omega \\
\grad u\cdot \bn &= g_N \quad \text{ on } \partial \Omega
\end{align*}
\end{block}

\begin{itemize}
\item other notation: $\grad u\cdot \bn = \frac{\partial u}{\partial n} = u_n$
\end{itemize}
\end{frame}


\begin{frame}{recall: the Sobolev spaces $H^1(\Omega)$ and $H_0^1(\Omega)$}

\small

\begin{definition}
    $$H^1(\Omega) = \left\{f\in L^2(\Omega) \,\Big|\, \begin{matrix} \text{the weak derivative } f' \text{ exists} \\ \text{and is in } L^2(\Omega)\end{matrix}\right\}$$
is a Hilbert space with inner product and norm:
	$$\ip{f}{g}_{H^1} = \int_\Omega f(x) g(x)\,dx + \int_\Omega \grad f(x)\cdot \grad g(x)\,dx, \qquad \|f\|_{H^1} = \sqrt{\ip{f}{f}_{H^1}}$$
\end{definition}

\begin{definition}
    $$H_0^1(\Omega) = \overline{C_c^1(\Omega)},$$
(closure is in $H^1$ norm, and $H_0^1(\Omega)$ is a closed subspace of $H^1(\Omega)$)
\end{definition}
\end{frame}


\begin{frame}{recall: the weak forms of the Poisson problems}

\begin{itemize}
\item multiply strong form by $v$ and use IBP to derive the weak forms:
\begin{align*}
-\int_\Omega (\grad^2 u) v\,dx &= -\int_{\partial \Omega} (\grad u\cdot \bn) v\,ds +\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx
\end{align*}
\end{itemize}

\begin{block}{weak formulation of the Poisson equation}
Suppose $f\in L^2(\Omega)$ and $g_N\in L^2(\partial\Omega)$ are given.  Solve one of these problems:
\begin{enumerate}
\item \alert{homogeneous Dirichlet:}  Find a solution $u\in H_0^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \quad \text{ for all test functions } v\in H_0^1(\Omega)$$
\item \alert{general Neumann:}  Find a solution $u\in H^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx + \int_{\partial\Omega} g_N v\,ds  \quad \text{ for all test functions } v\in H^1(\Omega)$$
\end{enumerate}
\end{block}
\end{frame}


\begin{frame}{recall: general Dirichlet boundary conditions}

\begin{itemize}
\item general Dirichlet conditions have an additional complication
\end{itemize}

\begin{block}{weak form: \alert{general Dirichlet}}
Suppose $f\in L^2(\Omega)$ and $g_D\in C^1(\partial\Omega)$ are given.  Let $u_D \in H^1(\Omega) \cap C(\bar\Omega)$ be any function, defined on the whole domain $\Omega$, so that $u_D=g_D$ along $\partial\Omega$.  Find a solution $u\in H_0^1(\Omega)$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx - \int_\Omega \grad u_D \cdot \grad v\,dx$$
for all test functions $v\in H_0^1(\Omega)$.
\end{block}

\begin{itemize}
\item the actually solution to the problem is $u+u_D$
\item even more generally, ``mixed'' conditions could partition the boundary: $\partial \Omega = \partial_D \Omega \cup \partial_N \Omega$
\end{itemize}

\end{frame}

\begin{frame}{recall: real Fourier series}

\small

\begin{lemma}  The Fourier series $g(x) = \frac{1}{\sqrt{2\pi}} \sum_{n\in\ZZ} c_n e^{inx}$ is a real-valued function if and only if $c_0\in\RR$ and $c_{-n} = \overline{c_n}$.
\end{lemma}

\begin{corollary}  If $g(x)$ is a real-valued function, and if for $n\ge 0$ we write $c_n = \frac{\sqrt{2\pi}}{2} (a_n-i b_n)$, with $a_n,b_n\in\RR$, then \, $\ds g(x) = \frac{a_0}{2} + \sum_{n=1}^\infty a_n \cos(n\theta) + b_n \sin(n\theta)$.
\end{corollary}

\footnotesize
\noindent \emph{Proof.} Write $g(x) = \frac{1}{\sqrt{2\pi}} c_0 + \frac{1}{\sqrt{2\pi}} \sum_{n=1}^\infty \left(c_n e^{inx} + c_{-n} e^{-inx}\right)$.  Then use the lemma, giving
	$$g(x) = \frac{1}{\sqrt{2\pi}} c_0 + \frac{1}{\sqrt{2\pi}} \sum_{n=1}^\infty 2 \Re \left(c_n e^{inx}\right)$$
Substitute $c_n = \frac{\sqrt{2\pi}}{2} (a_n-i b_n)$ for $n\ge 0$, and simplify, to get the result.\qed
\small

\begin{lemma}
    $$a_n = \frac{1}{\pi} \int_{-\pi}^\pi g(x)\cos(nx)\,dx, \qquad b_n = \frac{1}{\pi} \int_{-\pi}^\pi g(x)\sin(nx)\,dx$$
\end{lemma}
\end{frame}


\begin{frame}{harmonic functions on the disk}

\begin{itemize}
\item on the other hand, recall that the real and imaginary parts of any complex-analytic function on the disk are harmonic
\item for example, the monomials $z^n$, $n\ge 0$, are analytic
\item note $z=r e^{i\theta}$ in polar coordinates
\item the real and imaginary parts of $z^n$ are harmonic:
    $$z^n=r^n e^{in\theta}=r^n(\cos(n\theta) + i \sin(n\theta))$$
\end{itemize}

\begin{definition} for $n\ge 0$, define these \alert{harmonic functions} on $D$:
$$\phi_n(r,\theta) = r^n \cos(n\theta), \quad \psi_n(r,\theta) = r^n \sin(n\theta)$$
\end{definition}
\end{frame}


\begin{frame}{harmonic functions on the disk}

\begin{lemma} $\phi_n(r,\theta) = r^n \cos(n\theta)$ and $\psi_n(r,\theta) = r^n \sin(n\theta)$ are harmonic.  \end{lemma}

\footnotesize
\noindent \emph{Proof.}  In polar coordinates, $\alert{\grad^2 u = u_{rr} + \frac{1}{r} u_r + \frac{1}{r^2} u_{\theta\theta}}$.  Then for $u=\phi_n$,
\begin{align*}
\grad^2 u &= n (n-1) r^{n-2} \cos(n\theta) + \frac{1}{r} n r^{n-1} \cos(n\theta) + \frac{1}{r^2} (-n^2 r^n \cos(n\theta)) \\
  &= \Big[n (n-1) + n - n^2\Big] r^{n-2} \cos(n\theta) = 0 r^{n-2} \cos(n\theta) = 0
\end{align*}
Similarly for $u=\psi_n$. \hfill \qed

\begin{frame}{Laplace's equation (Dirichlet's problem) on a disk}

\begin{itemize}
\item combine these ideas to solve Laplace's equation $\grad^2 u = 0$ on the disk
\item start with Fourier series for $g\in L^2(\partial D)$; for $x\in (-\pi,\pi]$:
    $$g(x) = \frac{a_0}{2} + \sum_{n=1}^\infty a_n \cos(nx) + b_n \sin(nx)$$
\item use the coefficients to define a function on $\bar D$; for $0\le r \le 1$, $\theta \in (-\pi,\pi]$:
\begin{align*}
u(r,\theta) &= \frac{a_0}{2} + \sum_{n=1}^\infty a_n r^n \cos(n\theta) + b_n r^n \sin(n\theta) \quad \text{ for } (r,\theta) \in \bar D
\end{align*}
\end{itemize}

\vspace{-3mm}
\begin{theorem}  $u$ is continuous on $D$, $\grad^2 u=0$ within $D$, and $\ds \lim_{r\to 1^-} u(r,\theta) = g(\theta)$ with convergence in $L^2$.
\end{theorem}

\footnotesize
\noindent \emph{Proof.}  The series converges uniformly on any compact $K\subset D$, and the terms are continuous, so $u$ is continuous on $D$.  Compute $\|u(r,\cdot) - g(\cdot)\|_{L^2}$ and show it goes to 0 as $r\to 1$. \hfill \qed
\end{frame}

\begin{frame}{Poisson on a disk}

\begin{itemize}
\item now start from series expansion of source function:
$$f(r,\theta) = \sum_{n=1}^\infty a_{0n}\, \phi_{0n}(r,\theta) + \sum_{m,n=1}^\infty \big(a_{mn}\, \phi_{mn}(r,\theta) + b_{mn}\, \psi_{mn}(r,\theta)\big)$$
\item solve $-\grad^2 u = f$ using the eigenfunction property $\grad^2 \phi_{mn} = - \lambda_{mn}^2 \phi_{mn}$
\end{itemize}

\begin{theorem}  Suppose $f\in L^2(D)$ is expanded in $\phi_{mn}$, $\psi_{mn}$ as on the last slide, with convergence in $L^2$.  Define
\begin{equation*}
u(r,\theta) = \sum_{n=1}^\infty \frac{a_{0n}}{\lambda_{0n}^2}\, \phi_{0n}(r,\theta) + \sum_{m,n=1}^\infty \frac{1}{\lambda_{mn}^2} \big(a_{mn}\, \phi_{mn}(r,\theta) + b_{mn}\, \psi_{mn}(r,\theta)\big).
\end{equation*}
Then $u\in H_0^1(D) \cap C^2(D)$ solves the both the strong form and the weak form of the Poisson equation
    $$-\grad^2 u = f$$
\end{theorem}

\begin{frame}{spectral expansions for Poisson equation}

\begin{itemize}
\item we can combine the above techniques to solve the general Dirichlet problem for the Poisson equation on the disk:
   	$$-\grad^2 u = f\,\,\text{ on } D, \quad u=g \,\,\text{ on } \partial D$$

   \begin{itemize}
   \item[$\circ$] \emph{linearity} makes this possible   
   \item[$\circ$] method: (i) solve $-\grad^2 u_1 = 0$ and $u_1|_{\partial D}=g$ from Fourier series expansion of $g$, (ii) solve $-\grad^2 u_2 = f$ and $u_2|_{\partial D}=0$ from eigenfunction expansion of $f$, and (iii) then $u=u_1+u_2$

\item \dots only possible when the domain $\Omega$ has lots of symmetry, like the disk

   \item[$\circ$] also called \alert{separation of variables} (for PDEs)
   \item[$\circ$] essentially the domain has to be \emph{rectangular} in the preferred coordinates
   \item[$\circ$] for example, $D \cong [0,1) \times (-\pi,\pi]$ in the $r,\theta$ variables
   \end{itemize}
\end{itemize}

\section{Poisson equation well-posedness}

\begin{frame}{well-posedness of math problems}

\begin{definition}[Hadamard, 1902]
a mathematical problem is \alert{well-posed} if the solution satisfies the following properties:
\begin{enumerate}
\item it exists
\item it is unique
\item its value changes continuously as the data of the problem changes
\end{enumerate}
\end{definition}

\small
\begin{block}<2>{Example: finite-dimensional linear systems}
Suppose $A\in \CC^{m\times m}$ is invertible and $\bb\in\CC^m$.  The problem of solving \only<1>{$A\bx = \bb$}\only<2>{$\boxed{A\bx = \bb}$} for

$\bx\in\CC^m$ is well posed because
\begin{enumerate}
\item exists: $\bx = A^{-1}\bb$
\item unique: if $A\bx_1=\bb$ and $A\bx_2=\bb$ then $A(\bx_1-\bx_2)=\bzero$ so $\bx_1=\bx_2$
\item continuous in the data $\bb$: if $A\bx_1=\bb_1$ and $A\bx_2=\bb_2$ then

\quad $\|\bx_1-\bx_2\| = \|A^{-1}\bb_1 - A^{-1}\bb_2\| \le \|A^{-1}\| \|\bb_1 - \bb_2\|$
\end{enumerate}
\end{block}
\end{frame}


\begin{frame}{are Poisson problems well-posed?}

\begin{itemize}
\item can we prove that a boundary-value problem for Poisson is well-posed?
\item solving it in particular cases, e.g.~for the disk, will not help generally
\item recall: two important theorems which will help
\end{itemize}

\begin{theorem}[Poincar\'e-Friedrichs inequality]
Let $\Omega\subset \RR^d$ be an open, bounded domain with Lipschitz boundary, or an open set which is bounded in one direction.  There is $C>0$, depending on $\Omega$, so that
	$$\|f\|_{L^2} \le C \|\grad f\|_{L^2} \quad \text{for all } f\in H_0^1(\Omega).$$
\end{theorem}

\begin{theorem}[Riesz-Fr\'echet representation theorem]
If $(V,\ip{\cdot}{\cdot})$ is a (real) Hilbert space, and if $\ell \in V'$ is a bounded linear functional, then there exists $u\in V$ so that
	$$\ip{u}{v} = \ell(v) \quad \text{for all } v\in V$$
\end{theorem}
\end{frame}


\begin{frame}{Riesz representation route to well-posedness}

\begin{theorem}
The weak-form homogeneous Dirichlet problem, for $u\in H_0^1(\Omega)$ such that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \quad \text{ for all } v\in H_0^1(\Omega),$$
\alert{is well-posed}.
\end{theorem}

\small
\emph{Proof.} Let $\iip{u}{v} = \int_\Omega \grad u \cdot \grad v\,dx$.  We show that this is also an inner product on $H_0^1(\Omega)$.\footnote{The usual inner product is \, $\ip{u}{v} = \int_\Omega u v\,dx + \int_\Omega \grad u \cdot \grad v\,dx$.}  It is clearly nonnegative, symmetric, and bilinear, and furthermore it is nondegenerate: $\iip{u}{u}=0$ implies
	$$\|u\|_{H^1}^2 = \int_\Omega u^2\,dx + \int_\Omega |\grad u|^2\,dx \le (C^2+1)\int_\Omega |\grad u|^2\,dx = (C^2+1) \iip{u}{u} = 0,$$
by the Poincar\'e-Friedrichs inequality, so $u=0$.  This also shows that $\iip{\cdot}{\cdot}$ and $\ip{\cdot}{\cdot}$ give equivalent norms.  Thus $(H_0^1(\Omega),\iip{\cdot}{\cdot})$ is a real Hilbert space.
\end{frame}


\begin{frame}{proof of Riesz representation route 2}

\small
\emph{Proof, continued.} 
On the other hand, the right-hand side of the weak form defines a linear functional, $\ell:H_0^1(\Omega)\to \RR$, $\ell(v)=\int_\Omega fv\,dx$.  This is bounded by Cauchy-Schwarz,
    $$|\ell(v)| \le \int_\Omega |fv|\,dx \le \|f\|_{L^2}\|v\|_{L^2} \le \|f\|_{L^2}\|v\|_{H^1},$$
so $\ell \in H_0^1(\Omega)'$.  By the Riesz-Fr\'echet representation theorem there is a unique $u\in H_0^1(\Omega)$ so that
   $$\iip{u}{v} = \ell(v).$$
This exactly says that a solution to the weak form problem exists, and that is it unique.

\medskip
Finally, to show that the solution is continuous with respect to the data $f$, observe that the solution process is a well-defined linear map.  That is, there is a linear map $T:L^2(\Omega) \to H_0^1(\Omega)$ defined by\footnote{Conceptually, $T=(-\grad^2)^{-1}$.  But boundary conditions are involved!}
	$$T f = u$$
Recall that a linear map is continuous if and only if it is bounded.  We will show that
    $$\|u\|_{H^1} = \|T f\|_{H^1} \le c \|f\|_{L^2}.$$
\end{frame}


\begin{frame}{proof of Riesz representation route 3}

\small
\emph{Proof, continued.}
To show that $T$ is bounded we substitute $u$ for $v$ in the weak form, use a previous calculation, and apply Cauchy-Schwarz:
\begin{align*}
\|u\|_{H^1}^2 &\le (C^2+1) \iip{u}{u} = (C^2+1) \int_\Omega \grad u \cdot \grad u\,dx \\
  &= (C^2+1) \int_\Omega f u\,dx \le (C^2+1) \|f\|_{L^2} \|u\|_{L^2} \le (C^2+1) \|f\|_{L^2} \|u\|_{H^1}
\end{align*}
Now divide by the norm of $u$ to get a bound:
    $$\|u\|_{H^1} \le (C^2+1) \|f\|_{L^2}.$$
This shows that the solution map $Tf=u$ is continuous, and concludes the proof of well-posedness. \qed

\bigskip
\bigskip
\begin{itemize}
\item a bound like $\|u\|_{H^1} \le c \|f\|_{L^2}$ is often called an \alert{\emph{a priori} bound}
\item an \emph{a priori} bound is sufficient to conclude continuity (w.r.t.~$f$) for \emph{linear} problems
\end{itemize}
\end{frame}


\begin{frame}{the Neumann problem is \emph{not} well-posed}

\begin{itemize}
\item recall: the general weak-form Neumann problem is to find $u\in H^1(\Omega)$ s.t.
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx + \int_{\partial\Omega} g_N v\,ds  \quad \text{ for all } v\in H^1(\Omega)$$

    \begin{itemize}
    \item[$\circ$] strong form:\,\, $- \grad^2 u = f$ on $\Omega$,\, $\grad u\cdot \bn = g_N$ on $\partial \Omega$
    \end{itemize}
\item this has a uniqueness failure!
\end{itemize}

\begin{lemma}
If this problem has a solution then it has infinitely-many solutions.
\end{lemma}

\small
\emph{Proof.}  Let $u\in H^1(\Omega)$ be a solution and consider any $C\in\RR$.  Then $\tilde u(x)=u(x)+C$ is also in $H^1(\Omega)$, with the same gradient, and thus it satisfies the same weak form. \hfill \qed
\end{frame}


\begin{frame}{well-posed modifications of the Neumann problem}

\begin{itemize}
\item there are two well-known ways to fix this uniqueness issue
\item proofs for this theorem are skipped, but
    \begin{itemize}
    \item[$\circ$] note the strong form for \itemnum{1}:\,\, $\boxed{- \grad^2 u +\alpha u = f}$ on $\Omega$,\, $\grad u\cdot \bn = g_N$ on $\partial \Omega$
    \item[$\circ$] on Assignment 8: prove \itemnum{1} via the Riesz-Fr\'echet representation theorem
    \end{itemize}
\end{itemize}

\small
\begin{theorem}
Suppose $\Omega\subset \RR^2$ has a Lipschitz or $C^1$ boundary, $f\in L^2(\Omega)$, and $g_N\in L^2(\partial D)$.  The following weak-form Neumann-type problems for $u\in H^1(\Omega)$ \alert{are well-posed}:
\begin{enumerate}
\item \alert{Neumann problem for the Helmholtz equation}: \label{helm} Suppose $\alpha>0$.
    $$\int_\Omega \grad u \cdot \grad v\,dx + \int_\Omega \alpha u v\,dx = \int_\Omega f v\,dx + \int_{\partial\Omega} g_N v\,ds  \quad \text{ for all } v\in H^1(\Omega)$$
\item \alert{Neumann problem with auxiliary condition}:
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx + \int_{\partial\Omega} g_N v\,ds  \quad \text{ for all } v\in H^1(\Omega)$$
and so that $\int_\Omega u\,dx = 0$, equivalently $\ip{u}{c}=0$ for all constant functions $c$.
\end{enumerate}
\end{theorem}

\end{frame}

\begin{frame}{characterization theorem: minimization $\iff$ weak form}

\begin{itemize}
\item this idea is clearest in an abstract formulation
\end{itemize}

\begin{theorem}[characterization theorem]
Let $V$ be a real vector space.  Suppose
    $$a:V\times V\to \RR$$
is a symmetric and nonnegative ($a(v,v)\ge 0$) bilinear function.  Suppose $\ell:V\to\RR$ is a linear functional.  Define the quadratic objective function
	$$J(v) = \frac{1}{2} a(v,v) - \ell(v).$$
Then $u\in V$ is a minimizer of $J$ if and only if
	$$a(u,v) = \ell(v) \quad \text{ for all } v\in V.$$
\end{theorem}

\begin{itemize}
\item this theorem does \emph{not} assert existence or uniqueness; it \emph{characterizes} $u$
\item this theorem does \emph{not} require $V$ to be complete
\item this theorem does \emph{not} require the bilinear form $a$ to be positive definite
\end{itemize}
\end{frame}


\begin{frame}{proof of the characterization theorem 1}

\emph{Proof.}  Fix $u,v\in V$ and suppose $t\in \RR$.  Define
\begin{align*}
g(t) &= J(u+tv) = \frac{1}{2} a(u+tv,u+tv) - \ell(u+tv) \\
   &= \frac{1}{2} \Big(a(u,u) + 2 t\, a(u,v) + t^2 a(v,v)\Big) - \ell(u) - t \ell(v) \\
   &= J(u) + t\Big[a(u,v) - \ell(v)\Big] + \frac{1}{2} t^2 a(v,v).
\end{align*}

\medskip
Now if $u$ solves the weak form ($a(u,v) - \ell(v) =0$) then, using $t=1$ and nonnegativity of $a$,
	$$J(u+v) = g(1) = J(u) + \frac{1}{2} a(v,v) \ge J(u).$$
This holds for any $v\in V$, so $u$ is a minimizer of $J$.
\end{frame}


\begin{frame}{proof of the characterization theorem 1}

\emph{Proof, cont.}

\medskip
Conversely, suppose $J$ is minimized by $u$.  By assumption, the real function $g(t) = J(u+tv)$ has a minimum at $t=0$.  However, we have written $g$ as a quadratic polynomial in $t$, with derivative
	$$g'(t) = a(u,v) - \ell(v) + t\, a(v,v).$$
Since $u$ is a minimizer, $g'(0)=0$, which shows that $a(u,v)-\ell(v)=0$, so $u$ satisfies the weak form. \hfill \qed
\end{frame}


\begin{frame}{characterization, for the Poisson problem}

\begin{itemize}
\item example: the homogeneous Dirichlet problem $-\grad^2 u = f$, $u\big|_{\partial\Omega} = 0$
\item here: \, $V=H_0^1(\Omega)$, \, $a(u,v)=\int_\Omega \grad u \cdot \grad v\,dx$, \, $\ell(v)=\int_\Omega f v\,dx$
\end{itemize}

\begin{corollary}
The element $u\in H_0^1(\Omega)$ solves the weak-form problem
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \quad \text{ for all } v\in H_0^1(\Omega),$$
if and only if it is a minimizer of the \alert{energy functional} $J:H_0^1(\Omega) \to \RR$:
	$$u \quad \stackrel{\min}{\leftarrow} \quad J(v) = \frac{1}{2} \int_\Omega |\grad v|^2\,dx - \int_\Omega f v\,dx.$$
\end{corollary}

\begin{itemize}
\item again, we are \emph{not} (here) asserting existence of $u$
\item in the loaded elastic membrane application of the Poisson equation, $J(v)$ is the sum of the elastic ($\frac{1}{2} \int_\Omega |\grad v|^2\,dx$) and the gravitational/load ($- \int_\Omega f v\,dx$) potential energy
\end{itemize}
\end{frame}


\begin{frame}{minimization route to well-posedness}

{\footnotesize
\begin{itemize}
\item on a Hilbert space, if the bilinear form satisfies certain inequalities, then the minimizer exists and is unique
\end{itemize}
}

\begin{theorem}[existence of a minimizer]
Let $H$ be a Hilbert space.  Suppose $a:H\times H\to \RR$ is a symmetric bilinear function.  Assume it is \alert{\emph{coercive}} in the sense that there is $\alpha>0$ so that
	$$a(v,v) \ge \alpha \|v\|^2 \quad \forall v \in H.$$
Assume it is \alert{\emph{continuous}} in the sense that there is $C\ge 0$ so that
	$$|a(u,v)| \le C \|u\|\|v\| \quad \forall v \in H.$$
Suppose $\ell\in H'$.  Then there exists $u\in H$ which is the unique minimizer of $J$:
	$$u \quad \stackrel{\min}{\leftarrow} \quad J(v) = \frac{1}{2} a(v,v) - \ell(v).$$
\end{theorem}

{\footnotesize
\begin{itemize}
\item coercivity is also called (uniform) \alert{\emph{ellipticity}}
\item Braess (2007) \emph{Finite Elements}, calls this is a Lax-Milgram theorem.  I am not sure if that is right or wrong historically, but the usual Lax-Milgram theorem is \emph{not} about minimization.
\end{itemize}
}
\end{frame}


\begin{frame}{proof of the minimizer-existence theorem 1}

\emph{Proof.}  By coercivity and the definition of the $H'$ norm,
\begin{align*}
J(v) &= \frac{1}{2} a(v,v) - \ell(v) \ge \frac{\alpha}{2} \|v\|_H^2 - \|\ell\|_{H'} \|v\|_H \\
&= \frac{1}{2\alpha} \left(\alpha \|v\|_H - \|\ell\|_{H'}\right)^2 - \frac{\|\ell\|_{H'}^2}{2\alpha} \\
&\ge - \frac{\|\ell\|_{H'}^2}{2\alpha}.
\end{align*}
This shows $J$ is bounded below.  Thus
	$$\mu = \inf \left\{J(v)\,:\,v\in V\right\}$$
exists as a real number.

\medskip
We will show that there is a $u\in V$ so that $J(u)=\mu$, namely that a minimizer achieves the infimum.
\end{frame}


\begin{frame}{proof of the minimizer-existence theorem 2}

\emph{Proof, cont.}

\medskip
Suppose $(v_n)$ is a sequence in $V$ such that $J(v_n) \to \mu$, that is, it is a minimizing sequence.  Then\footnote{We used $J\left(\frac{v_n+v_m}{2}\right)\ge \mu$ in the last step.  This argument generalizes to convex $C\subset H$.}
\begin{align*}
\alpha \|v_n-v_m\|_H^2 &\le a(v_n-v_m,v_n-v_m) \\
    &= 2 a(v_n,v_n) + 2 a(v_m,v_m) - a(v_n+v_m,v_n+v_m) \phantom{\Big|} \\
    &= 2 a(v_n,v_n) - 4\ell(v_n) + 2 a(v_m,v_m) - 4\ell(v_n) \\
    &\qquad - 4 a\left(\frac{v_n+v_m}{2},\frac{v_n+v_m}{2}\right) + 8\ell\left(\frac{v_n+v_m}{2}\right) \\
    &= 4 J(v_n) + 4 J(v_m) - 8 J\left(\frac{v_n+v_m}{2}\right) \\
    &\le 4 J(v_n) + 4 J(v_m) - 8 \mu.
\end{align*}
But since $J(v_n) \searrow \mu$ and $J(v_m) \searrow \mu$, this inequality shows $\|v_n - v_m\|\to 0$, so $(v_n)$ is Cauchy.
\end{frame}


\begin{frame}{proof of the minimizer-existence theorem 3}

\emph{Proof cont.}

\medskip
Since $H$ is complete,\footnote{Here the argument would need $C\subset H$ to be closed, in addition to convex.} $u=\lim_{n\to\infty} v_n \in H$ exists.  The continuity of $J$ at $u$, which follows from the continuity of $a$ and $\ell$, now shows
	$$J(u) = J\left(\lim_{n\to\infty} v_n\right) = \lim_{n\to\infty} J(v_n) = \mu,$$
so a minimizer exists.

\medskip
Now suppose that $u_1,u_2$ are both minimizers, so $J(u_i)=\mu$.  The previous calculation\footnote{Here we are seeing that $J$ is \emph{strictly} convex, thus the minimizer is unique.} applied to $u_1$ and $u_2$ gives zero exactly:
    $$\alpha \|u_1-u_2\|^2 \le 4 J(u_1) + 4 J(u_2) - 8 \mu = 0.$$
Thus $u_1=u_2$, so the solution is unique. \hfill \qed
\end{frame}

\begin{itemize}

\begin{corollary}
The solution $u\in H_0^1(\Omega)$ of the weak-form problem
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \quad \text{ for all } v\in H_0^1(\Omega)$$
exists and is unique.
\end{corollary}

\emph{Proof.}  Apply the characterization theorem to turn the weak form into a minimization.  Show coercivity and continuity of $a(u,v) = \int_\Omega \grad u \cdot \grad v\,dx$.  (Coercivity will need the Poincar'e-Friedrichs inequality.)  Then apply the last theorem to show the minimizer exists and is unique. \hfill \qed
}
