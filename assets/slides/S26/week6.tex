\section{measurable subsets of $\RR^1$, and Lebesgue measure}

\begin{frame}{intervals and their lengths}

\begin{itemize}
\item in Chapter 3 our textbook defines \emph{measurable subset} (of $\RR^n$), and then \emph{simple function} (from $\RR^n$ to $\RR$)
\item I will give the definitions for $\RR^1$ in these slides
\end{itemize}

\begin{definition}
\begin{itemize}
\item for real numbers $a\le b$, a \emph{finite interval} $I\subset \RR^1$ is either
    $$(a,b) \text{ or } (a,b] \text{ or } [a,b) \text{ or } [a,b]$$ 
\item the \emph{length} of a (finite) interval is
    $$m(I) = b - a < \infty$$
\end{itemize}
\end{definition}

\begin{itemize}
\item singleton sets, and the empty set, are finite intervals with length zero
   \begin{itemize}
   \item[$\circ$] $\{a\} = [a,a]$ and $\emptyset = (a,a)$
   \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{outer measure}

\begin{definition}
for any subset $A\subset \RR^1$, the \emph{outer measure} is
	$$m^*(A) = \inf\, \left\{\sum_{k=1}^\infty m(I_k)\,:\,A \subset \bigcup_{k=1}^\infty I_k\right\}$$
where the infimum is over countable unions of (finite) intervals $I_k$ covering $A$
\end{definition}

\begin{itemize}
\item $m^*(A)$ is defined for \emph{any} subset $A\subset \RR^1$
\item thus $m^* : 2^{(\RR^1)} \to [0,+\infty]$
   \begin{itemize}
   \item[$\circ$] the \emph{power set} of a set $S$ is \, $2^S = \left\{A \,:\, A\subset S\right\}$
   \end{itemize}
\item $m^*(A)=+\infty$ is possible
\end{itemize}
\end{frame}


\begin{frame}{outer measure: examples}
\begin{itemize}
\item[Ex 1] if $I$ is any interval then $m^*(I)$ is the length of $I$
   \begin{itemize}
   \item[$\circ$] $m^*(\RR)=+\infty$, and likewise for half-infinite intervals like $[a,+\infty)$
   \end{itemize}
\item[Ex 2] if $A$ is a finite or countable set then $m^*(A)=0$

\end{proof}
\item[Ex 3] $m^*(\QQ)=0$
\item[Ex 4] $m^*([0,1]\setminus\QQ)=1$
\item[Ex 5] $m^*(C)=0$ where $C \subset [0,1]$ is the \emph{Cantor middle-thirds set}  \hfill [\emph{picture}]

\end{itemize}
\end{frame}


\begin{frame}{measure zero}

\begin{definition}
a subset $A\subset \RR^1$ has \emph{measure zero} if $m^*(A)=0$
\end{definition}

\begin{lemma}
$A\subset \RR^1$ has measure zero if and only if for every $\eps>0$ there is a countable covering of $A$ by intervals of total length less than $\eps$:
	$$\sum_{k=1}^\infty m(I_k) < \eps \qquad \text{ where } A \subset \bigcup_{k=1}^\infty I_k$$
\end{lemma}

{\small \noindent \emph{Proof.}
Just think through the definition of $m^*(A)=0$.  \qed
}

\bigskip
\begin{itemize}
\item every subset of a measure zero set has measure zero
\item every finite or countable set has measure zero
\item the Cantor set $C\subset[0,1]$, though it has uncountable cardinality, has measure zero
\end{itemize}
\end{frame}


\begin{frame}{symmetric difference of sets}

\begin{definition}
given two sets $A,B$, their \emph{symmetric difference} is the set
	$$S(A,B) = (A \setminus B) \cup (B \setminus A) = (A \cup B) \setminus (A\cap B)$$
\end{definition}

\begin{frame}{Lebesgue measurable subsets of $\RR^1$}

\begin{definition}
$\cM$ is the collection of \emph{Lebesgue measurable sets}
\end{definition}

\begin{itemize}
\item every interval is measurable: $I \in \cM$
\item every finite union of intervals is measurable: $\cE \subset \cM$
\item every set of measure zero is measurable
\item every measurable set can be well-approximated by a countable, disjoint union of intervals:
	$$A \in \cM \iff \forall \eps>0\, \exists\, B=\bigcup_{k=1}^\infty I_k \text{ so that } m^*(S(A,B))<\eps$$
\item<2> \alert{there exist $A\subset \RR^1$ that are not Lebesgue measurable}:
   $$\cM \subsetneq 2^{(\RR^1)}$$
\item<2> I will be covering this Chapter 3 material somewhat more carefully in the next few days
\end{itemize}
\end{frame}


\begin{frame}{two very useful lemmas}

\begin{lemma}
$G \subset \RR^1$ is open if and only if it is a countable, disjoint union of open intervals:
	$$G = \bigcup_{k=1}^\infty (a_k, b_k) \qquad \text{ where } a_k,b_k \in [-\infty,+\infty]$$
\end{lemma}

\begin{itemize}
\item thus $m^*(G) = \sum_{k} b_k - a_k$ for open sets $G$
\end{itemize}

\begin{lemma}
a measurable set can be squeezed between a closed and an open set:
	$$A \in \cM \qquad \iff \qquad \begin{matrix} \forall \eps>0 \, \exists E \text{ closed } \, \exists G \text{ open } \text{ s.t. } \hspace{10mm} \\ E \subset A \subset G\, \text{ and }\, m^*(G\setminus E)<\eps \end{matrix}$$
\end{lemma}

\vspace{13mm}
\end{frame}


\begin{frame}{but what is ``Lebesgue measure''?}

\begin{definition}
\emph{Lebesgue measure} is the function $m:\cM \to [0,+\infty]$ defined as the outer measure restricted to the measurable sets:
\begin{align*}
m : \cM &\to [0,+\infty] \\
    A &\mapsto m^*(A)
\end{align*}
\end{definition}

\begin{itemize}
\item what's the key property of a measure?
\end{itemize}

\begin{lemma}
$m$ is \alert{countably additive}, that is, if $A_k\in \cM$ are pairwise disjoint ($A_j \cap A_k = \emptyset$ for any $j\ne k$) then
	$$m\left(\bigcup_{k=1}^\infty A_k\right) = \sum_{k=1}^\infty m(A_k)$$
\end{lemma}
\end{frame}

\section{simple functions}

\begin{frame}{simple functions}

\begin{definition}
the \emph{indicator function} for the set $A\subset \RR^1$ is
	$$\one_A(x) = \begin{cases} 1, & x\in A \\ 0, & x\notin A \end{cases}$$
\end{definition}

\begin{itemize}
\item also denoted $\chi_A(x)$
\end{itemize}

\begin{definition}
a \emph{simple function}, $s:\RR^1\to\RR$, is a \alert{finite linear combination of indicator functions of measurable sets} $E_k \in \cM$, using coefficients $c_k\in\RR$:
	$$s(x) = \sum_{k=1}^N c_k \one_{E_k}(x)$$
\end{definition}
\end{frame}


\begin{frame}{integrals of simple functions}

\begin{itemize}
\item the integral of a simple function \, $\ds s(x) = \sum_{k=1}^N c_k \one_{E_k}(x)$ \, is obvious?
\end{itemize}

\begin{definition}
if $s$ is a measurable simple function then its \emph{Lebesque integral} is
	$$\int_\RR s(x)\,dx = \sum_{k=1}^N c_k m(E_k),$$
assuming for simplicity that $\sum_k m(E_k)<\infty$
\end{definition}

\end{frame}


\begin{frame}{the fundamental idea of Lebesgue integrals}

\begin{itemize}
\item idea: \alert{cut-up the \emph{range}} of $f:[a,b] \to \RR$, so as to approximate by a simple function, and integrate that simple function
   \begin{itemize}
   \item[$\circ$] $f$ has to be \emph{measurable} \hfill \alert{$\longleftarrow$ we will define!}
   \end{itemize}
\item the range is cut into $y_0<y_1<\dots<y_N$ to define the simple function:
	$$s(x) = \sum_{k=0}^{N-1} y_{k}\, \one_{f^{-1}([y_{k},y_{k+1}))}(x)$$

   \begin{itemize}
   \item[$\circ$] $E_k = f^{-1}([y_{k},y_{k+1}))$ are the measurable sets
   \end{itemize}
\end{itemize}

\begin{definition}[nearly the right definition]
for a measurable $f:[a,b] \to \RR$, the \emph{Lebesgue integral} is
    $$\int_a^b f(x)\,dx = \lim_{N\to \infty} \sum_{k=0}^{N-1} y_{k}\, m\left(f^{-1}([y_{k},y_{k+1}))\right)$$
\end{definition}

\begin{itemize}
\item the careful and correct definition first restricts to \emph{nonnegative} functions, and the uses ``$\sup$'' instead of the ``$\lim_{N\to\infty}$''
\end{itemize}
\end{frame}

\section{step functions ($=$ piecewise-constant on meshes)}

\begin{frame}{step functions}

\begin{itemize}
\item simple functions $\ds s(x) = \sum_{k=1}^N c_k \one_{E_k}(x)$ are as complicated as the sets $E_k$
    \begin{itemize}
    \item[$\circ$] measurable sets $E_k$ can be quite intricate, e.g.~Cantor sets
    \item[$\circ$] message: \alert{simple functions aren't actually simple}
    \end{itemize}
\item what if we restrict $E_k$ to be intervals?
\item given a 1D mesh $x_0<x_1<x_2<\dots<x_n$, define the (finite) intervals to be $I_k=[x_{k-1},x_k]$ or $[x_{k-1},x_k)$ or $(x_{k-1},x_k]$ or $(x_{k-1},x_k)$
\end{itemize}

\begin{definition}[equivalent forms]
\begin{itemize}
\item given $x_0<x_1<x_2<\dots<x_n$ and $c_k \in \RR$ a \emph{step function} is
     $$q(x) = \sum_{k=1}^N c_k \one_{I_k}(x)$$
\item a simple function is a \emph{step function} if each $E_k$ is a finite interval
\end{itemize}
\end{definition}
\end{frame}


\begin{frame}{step functions}

\begin{definition}
given the mesh $x_0<x_1<x_2<\dots<x_n$ and $c_k \in \RR$ a \emph{step function} is
     $$q(x) = \sum_{k=1}^N c_k \one_{I_k}(x),$$
where $I_k$ is an interval of the mesh
\end{definition}

\begin{frame}{the two integrals (rough comparison)}

\begin{itemize}
\item simple functions \alert{from meshing the range} $\leftrightarrow$ Lebesgue integrals
\item step functions \alert{from meshing the domain} $\leftrightarrow$ Riemann integrals
\end{itemize}

\begin{block}{integration concepts}
\begin{itemize}
\item approximate $f:[a,b] \to \RR$ by \alert{simple} functions, integrate those using the Lebesgue measure, and take a limit to get the \emph{Lebesgue integral}:
    $$\int_a^b f(x)\,dx = \lim_{N\to \infty} \sum_{k=0}^{N-1} y_{k}\, m\left(f^{-1}([y_{k},y_{k+1}))\right)$$
\item approximate $f:[a,b] \to \RR$ by \alert{step} functions, integrate those, and take a limit to get the \emph{Riemann integral}:
    $$\int_a^b f(x)\,dx = \lim_{N\to \infty} \sum_{k=0}^{N-1} f(x_k^*)\, (x_k-x_{k-1})$$
\end{itemize}
\end{block}
\end{frame}


\begin{frame}{density theorems (mostly $\RR^1$ case)}

\begin{theorem}
\begin{enumerate}
\item for every $f\in C([a,b])$ and $\eps>0$ there exists a step function $q$ such that $\|f-q\|_\infty < \infty$
\item the simple functions are dense in $L^p(X,\mu),\|\cdot\|_p$, for $1 \le p \le \infty$
\item the step functions are dense in $L^p(E,m),\|\cdot\|_p$, for $1 \le p < \infty$
    \begin{itemize}
    \item[$\circ$] here $E \subset \RR^1$ is measurable, e.g.~any interval
    \end{itemize}
\item the continuous functions $C([a,b])$ are dense in $L^p([a,b],m)$ for $1 \le p < \infty$
\end{enumerate}
\end{theorem}

\begin{itemize}
\item density of simple functions (item \itemnum{2}) in $L^p(X,\mu)$ is (essentially) the definition of the Lebesgue integral
    \begin{itemize}
    \item[$\circ$] this fact is completely general
    \end{itemize}
\item density of step functions $\implies$ density of simple functions
\item \itemnum{1} is a 19th century fact which follows from uniform continuity
    \begin{itemize}
    \item[$\circ$] I have already used this idea in proving the Riemann-Lebesgue lemma
    \end{itemize}
\end{itemize}
\end{frame}


\section{piecewise-linear functions}

\begin{frame}{piecewise-linear functions on 1D meshes}

\begin{itemize}
\item restrict the (finite) intervals to be closed: $I_k=[x_{k-1},x_k]$
\end{itemize}

\begin{definition}
\begin{itemize}
\item given a mesh $a=x_0<x_1<x_2<\dots<x_n=b$ and coefficients $c_k, d_k \in \RR$, a \emph{continuous piecewise-linear function} on $[a,b]$ is
     $$p(x) = \sum_{k=1}^N (c_k (x-x_{k-1}) + d_k) \, \one_{I_k}(x)$$
if $c_{k-1} (x_k-x_{k-1}) + d_{k-1} \stackrel{\ast}{=} d_k$ for $k=2,3,\dots,N$
\item a \emph{discontinuous piecewise-linear ``function''} is the same thing, but without requirement $\ast$
\end{itemize}
\end{definition}

\begin{frame}{observation about piecewise functions}

\begin{itemize}
\item I have already used piecewise functions many times in proofs and counter-examples
    \begin{itemize}
    \item[$\circ$] often these are ``hat functions'' or moving spikes etc.
    \item[$\circ$] the \alert{easiest constructable continuous functions} are often piecewise-linear
    \item[$\circ$]the \alert{easiest constructable discontinuous functions} are often piecewise-constant
    \end{itemize}
\item a piecewise-linear or piecewise-constant ($=$ step) function \alert{is defined using finitely-many real numbers}, so it can be stored on a computer
\item continuous piecewise-linear functions are foundational in applied and computational mathematics
    \begin{itemize}
    \item[$\circ$] generally: piecewise-polynomial functions
    \item[$\circ$] also finite Fourier series ($=$ trigonometric polynomials)
    \end{itemize}
\item simple and step functions are foundational in theory
    \begin{itemize}
    \item[$\circ$] in fact piecewise-constant functions are very useful in practice
    \end{itemize}
\end{itemize}
\end{frame}
