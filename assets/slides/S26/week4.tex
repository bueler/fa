\section{what did Fourier believe? (1822)}

\begin{frame}{Fourier's assertion}

\begin{block}{claim (Fourier, \emph{Th\'eorie Analytique de la Chaleur}, 1822)}
if $f\in C[0,\pi]$, and if we compute these coefficients
	$$c_n = \ip{f}{\phi_n} = \sqrt{\frac{2}{\pi}} \int_0^\pi f(x) \sin(nx)\,dx$$
for $n\in \NN$, then
	$$f(x) = \sum_{n=1}^\infty c_n\, \phi_n(x) = \sqrt{\frac{2}{\pi}} \sum_{n=1}^\infty c_n \sin(nx)$$
\end{block}

\begin{itemize}
\item the result is the \emph{Fourier sine series} (or \emph{expansion}) of $f$
\item actually Fourier claimed this for discontinuous functions too
   \begin{itemize}
   \item[$\circ$] he, and everybody else, agreed that the claim could not be exactly and generally true at the points of discontinuity of $f(x)$
   \item[$\circ$] observe that $c_n$ is an \emph{integral}, so only features of $f$ that change the integrals can affect the expansion
   \end{itemize}
\end{itemize}
\end{frame}


\begin{frame}{complex version is easier}

\begin{itemize}
\item Fourier used real numbers (to my knowledge)
\item we may replace
	$$C[0,\pi] \to C[-\pi,\pi] \qquad \text{and} \qquad \sin(nx) \to e^{inx},$$
and use the complex (sesquilinear) inner product
\end{itemize}

\begin{block}{Fourier's claim}
if $f\in C[-\pi,\pi]$, complex-valued, and if
	$$c_n = \ip{f}{\psi_n} = \frac{1}{\sqrt{2\pi}} \int_{-\pi}^\pi f(x) e^{-inx}\,dx$$
for $n\in\ZZ$, then
	$$f(x) = \sum_{n\in\ZZ} c_n\, \psi_n(x) = \frac{1}{\sqrt{2\pi}} \sum_{n\in\ZZ} c_n e^{inx}$$
\end{block}

\begin{itemize}
\item this is the \emph{Fourier series} of $f$, or the \emph{complex Fourier series}
\item for formulas from different references, be careful with where the $2\pi$ goes
\end{itemize}
\end{frame}

\begin{itemize}
\item consider the Fourier sine series for $f(x)=x$, using the real orthonormal set $\{\phi_n(x) = \sqrt{\frac{2}{\pi}} \sin(n x)\} \subset C[0,\pi]$
\item integrate to get coefficients:
\begin{align*}

&\phantom{x} \\
&\phantom{x} \hspace{80mm} = \frac{\sqrt{2\pi} (-1)^{n-1}}{n}
\end{align*}
\end{itemize}
\end{frame}

\section{what did Dirichlet prove? (1829)}

\begin{frame}{approximating $f$}

\begin{itemize}
\item again consider complex Fourier series for $f\in C[-\pi,\pi]$, namely
   $$c_n = \frac{1}{\sqrt{2\pi}} \int_{-\pi}^\pi f(x) e^{-inx}\,dx, \qquad f(x) = \frac{1}{\sqrt{2\pi}} \sum_{n\in\ZZ} c_n e^{inx}$$
\item combine into one formula and exchange limit processes:
\begin{align*}
f(x) &= \frac{1}{\sqrt{2\pi}} \sum_{n\in\ZZ} \left(\frac{1}{\sqrt{2\pi}} \int_{-\pi}^\pi f(y) e^{-iny}\,dy\right) e^{inx} \\
&= \int_{-\pi}^\pi \left(\frac{1}{2\pi} \sum_{n\in\ZZ} e^{-iny} e^{inx} \right) f(y)\,dy = \int_{-\pi}^\pi \left(\alert{\frac{1}{2\pi} \sum_{n\in\ZZ} e^{in(x-y)}}\right) f(y)\,dy
\end{align*}

    \begin{itemize}
    \item[$\circ$] exchange of integral and sum is justified if $\sum |c_n|<\infty$
    \end{itemize}

\medskip
\item if the \alert{inner sum} is made into a partial sum, then we are approximating $f$ by integrating it against a kernel
\end{itemize}
\end{frame}

\end{definition}

\begin{frame}{properties of the kernel}

{\footnotesize
\begin{definition}[Dirichlet's kernel]
$$D_m(x) = \frac{1}{2\pi} \sum_{n=-m}^m e^{inx} = \frac{1}{2\pi} + \frac{1}{\pi} \sum_{n=1}^m \cos(nx)$$
\end{definition}
}

\begin{lemma}[properties of the kernel]
\begin{enumerate}
\item $\ds \int_{-\pi}^\pi D_m(x)\,dx = 1$
\item $D_m(0) = \frac{m+1/2}{\pi}$, and $D_m(0) \to \infty$
\item $\ds D_m(x) = \frac{1}{2\pi} \frac{\sin((m+1/2)x)}{\sin(x/2)}$ \, for $x\ne 0$
\end{enumerate}
\end{lemma}

{\small
\noindent \emph{Proof.}  Properties \itemnum{1} and \itemnum{2} are easy from the sum definition of $D_m$.  Property \itemnum{3} is on the next slide. \hfill \qed
}
\end{frame}


\begin{frame}{Dirichlet's kernel as a sinc function}

Property \itemnum{3}: \, $\ds D_m(x) = \frac{1}{2\pi} \frac{\sin((m+1/2)x)}{\sin(x/2)}$

{\small
\noindent \emph{Proof.}  Apply knowledge of geometric series, then use $\sin\theta = (e^{i\theta}-e^{-i\theta})/(2i)$:
\begin{align*}
\sum_{n=-m}^m e^{inx} &= \sum_{n=-m}^m \left(e^{ix}\right)^n = e^{-imx} \sum_{n=0}^{2m} \left(e^{ix}\right)^n \\
&= e^{-imx} \frac{1 - (e^{ix})^{2m+1}}{1 - e^{ix}} = \frac{e^{-imx} - e^{i(m+1)x}}{1 - e^{ix}} \\
&= \frac{e^{-i(m+1/2)x} - e^{i(m+1/2)x}}{e^{-i(x/2)} - e^{i(x/2)}} \\
&= \frac{\sin((m+1/2)x)}{\sin(x/2)} \hspace{40mm} \qed
\end{align*}
}

\begin{itemize}
\item engineers who work with signals define\, $\ds \operatorname{sinc}(x) = \frac{\sin x}{x}$
\end{itemize}
\end{frame}


\begin{frame}{Dirichlet's theorem (1829)}

\begin{theorem}
If $f\in C[-\pi,\pi]$ is periodic, and if $f'(x)$ exists at $x\in[-\pi,\pi]$, then
    $$\lim_{m\to\infty} \int_{-\pi}^\pi D_m(x-y)\, f(y)\,dy = f(x),$$
and thus Fourier's claim is true:
{\small
$$f(x) = \frac{1}{\sqrt{2\pi}} \sum_{n\in\ZZ} c_n e^{inx} \quad \text{ for }\, c_n =\frac{1}{\sqrt{2\pi}} \int_{-\pi}^\pi f(x) e^{-inx}\,dx$$
}
\end{theorem}

\begin{itemize}
\item The proof is based on the \emph{Riemann-Lebesgue lemma} for continuous functions.  This version only requires the Riemann integral, and Dirichlet might have known it.  I will prove it, but after using it for the theorem.
\end{itemize}

\begin{lemma}
If $g \in C[a,b]$ then $\ds \lim_{n\to\infty} \int_a^b g(x) e^{inx}\,dx = 0$.
\end{lemma}
\end{frame}


\begin{frame}{Dirichlet's theorem (1829)}

{\small
\noindent \emph{Proof.}  By the periodicity of $D_m$ and $f$, and since $D_m$ is even, use $y=x+\xi$ to show
    $$\int_{-\pi}^\pi D_m(x-y)\, f(y)\,dy = \int_{-\pi}^\pi D_m(\xi)\, f(x+\xi)\,d\xi$$
By replacing $f(x+\xi)$ with $\tilde f(\xi)$, we may assume $x=0$ and $f'(0)$ exists.  Next we calculate using the proven properties of the kernel:
\begin{align*}
\int_{-\pi}^\pi D_m(\xi)\, f(\xi)\,d\xi &= \int_{-\pi}^\pi D_m(\xi)\, \left(f(\xi) - f(0)\right)\,d\xi + \int_{-\pi}^\pi D_m(\xi)\, f(0)\,d\xi \\
  &= f(0) + \frac{1}{2\pi} \int_{-\pi}^\pi \left(f(\xi) - f(0)\right)\,\frac{\sin((m+1/2)\xi)}{\sin(\xi/2)}\,d\xi
\end{align*}
However, $f'(0)$ exists, so by L'Hopital's rule,
	$$\lim_{\xi\to 0} \frac{f(\xi) - f(0)}{\sin(\xi/2)} = \lim_{\xi\to 0} \frac{f(\xi) - f(0)}{\xi} \frac{\xi}{\sin(\xi/2)} = 2 f'(0)$$
}
\end{frame}


\begin{frame}{Dirichlet's theorem (1829)}

{\small
\noindent \emph{Proof continued.}  So now we know that this function is continuous (removable discontinuity):
	$$h(x) = \begin{cases} \frac{f(x) - f(0)}{\sin(x/2)}, & x\ne 0 \\
	                       2f'(0), & x=0 \end{cases}$$
Thus by the Riemann-Lebesgue lemma:
\begin{align*}
\lim_{m\to\infty} \int_{-\pi}^\pi D_m(\xi)\, f(\xi)\,d\xi &= f(0) + \frac{1}{2\pi} \lim_{m\to\infty}  \int_{-\pi}^\pi h(\xi) \sin((m+1/2)\xi)\,d\xi \\
&= f(0) + 0 = f(0).  \hspace{30mm} \qed
\end{align*}
}
\end{frame}


\begin{frame}{Riemann-Lebesgue lemma for continuous functions}

\begin{lemma}
If $g \in C[a,b]$ then $\ds \lim_{n\to\infty} \int_a^b g(x) e^{inx}\,dx = 0$.
\end{lemma}

{\small
\noindent \emph{Proof.}  Let $\eps>0$.  Since $g$ is continuous on the compact set $[a,b]$, it is uniformly continuous, so there is $\delta>0$ such that $|x-y|<\delta \implies |g(x)-g(y)|<\eps/(2(b-a))$.  In particular we can define a mesh of points $\{x_i\}_{i=0}^M$, with spacing less than $\delta$, and $x_i^*$ at the midpoints of each $[x_{i-1},x_i]$, so that
	$$\tilde g(x) = \sum_{i=1}^M g(x_i^*) \one_{[x_{i-1},x_i)}(x)$$
is a uniform approximation of $g$: \quad $\ds \|\tilde g - g\|_\infty = \sup_{x\in [a,b]} |\tilde g(x) - g(x)| < \frac{\eps}{2(b-a)}$

picture:

\vspace{25mm}
}
\end{frame}


\begin{frame}{Riemann-Lebesgue lemma for continuous functions}

{\small
\noindent \emph{Proof continued.}  Now
\begin{align*}
\left|\int_a^b g(x) e^{inx}\,dx\right| &\le \left|\int_a^b (g(x) - \tilde g(x)) e^{inx}\,dx\right| + \left|\int_a^b \tilde g(x) e^{inx}\,dx\right| \\
&\le \|g-\tilde g\|_\infty \int_a^b |e^{inx}|\,dx + \left|\int_a^b \sum_{i=1}^M g(x_i^*) \one_{[x_{i-1},x_i)}(x) e^{inx}\,dx\right| \\
&\le \frac{\eps}{2(b-a)}(b-a) + \sum_{i=1}^M |g(x_i^*)| \left|\int_{x_{i-1}}^{x_i} e^{inx}\,dx\right|
\end{align*}
But each of the final integrals is bounded by the same constant:
    $$\left|\int_{x_{i-1}}^{x_i} e^{inx}\,dx\right| = \left|\frac{e^{i n x_i} - e^{i n x_{i-1}}}{in}\right| \le \frac{2}{n}$$
Choose $N$ so that $(2\|g\|_\infty M)/N < \eps/2$.  If $n\ge N$ then
  $$\left|\int_a^b g(x) e^{inx}\,dx\right| \le \frac{\eps}{2} + \frac{2}{n} \sum_{i=1}^M |g(x_i^*)| \le \frac{\eps}{2} + \frac{2}{N} \sum_{i=1}^M \|g\|_\infty < \frac{\eps}{2} +  \frac{\eps}{2} = \eps \hspace{5mm} \qed$$
}
\end{frame}


\section{what is actually true for Fourier series on $C[-\pi,\pi]$? (1966)}

\begin{frame}{does the Fourier series converge for any $f\in C[-\pi,\pi]$?}

\begin{itemize}
\item in 1873 Du Bois-Reymond constructed a continuous function for which the Fourier series did not converge at a point
    \begin{itemize}
    \item[$\circ$] in \href{https://www.mathcounterexamples.net/continuous-function-with-divergent-fourier-series/}{a search I found}: \quad $\ds f(x) = \sum_{k=1}^\infty \frac{1}{k^2} \sin\left(\frac{(2^{k^3}+1)|x|}{2}\right)$
    \item[$\circ$] the function is even and continuous (by

Weierstrass M-test)
    \item[$\circ$] it can be shown that the Fourier partial sums

are unbounded in a neighborhood of $x=0$
    \end{itemize}

\vspace{-13mm}

\hfill \includegraphics[width=0.35\textwidth]{figs/duboisreymond.png}

\medskip
\item later examples fail convergence on infinite sets of points
\item to state the full situation we need to define \emph{measure zero}
\end{itemize}
\end{frame}


\begin{frame}{sets of measure zero}

\begin{definition}
\begin{itemize}
\item a set $E\subset \RR$ has \emph{(Lebesgue) measure zero} if for any $\eps>0$ there exists a countable list of open intervals $I_n=(a_n,b_n)$ so that
    $$E \subset \bigcup_{n\in\NN} I_n \quad \text{and} \quad \sum_{n\in\NN} b_n-a_n < \eps$$
\item if some fact holds except on a set $E$ which has measure zero, we say that the fact holds \emph{almost everywhere}
\end{itemize}
\end{definition}

\begin{itemize}
\item note that any open subset of $\RR$ is a countable union of open intervals
\item more on measure zero in Chapter 3
\end{itemize}
\end{frame}


\begin{frame}{Carleson's and Katznelson's theorems}

\begin{theorem}[Carleson, 1966]
if $f\in C[-\pi,\pi]$, or even $f\in L^p[-\pi,\pi]$, then the Fourier series of $f$ converges pointwise almost everywhere:
	$$\lim_{m\to\infty} \sum_{n=-m}^m c_n e^{inx} = f(x) \quad \text{except on $E\subset [-\pi,\pi]$ of measure zero}$$
\end{theorem}

\begin{theorem}[Katznelson, 1966]
for any set $E\subset [-\pi,\pi]$ of measure zero there is a continuous periodic function $f\in C[-\pi,\pi]$ whose Fourier series diverges at all points of $E$
\end{theorem}
\end{frame}


\section{what is the clean $L^2$ version? ($\sim$1910)}

\begin{frame}{orthonormal sequences in inner product sets}

\begin{itemize}
\item the ideas of Hilbert and Riesz, circa 1910, are fundamentally \emph{simpler} than all the previous material in these slides
\item consider an abstract inner product vector space $V, \ip{\cdot}{\cdot}$
\end{itemize}

\begin{definition}
\begin{itemize}
\item vectors $v,w \in V$ are \emph{orthogonal} if $\ip{v}{w}=0$
\item a set $S$ is \emph{orthogonal} if $v,w\in S$ and $v\ne w$ implies $v,w$ are orthogonal
\item $(v_k)_{k=1}^\infty \subset V$ is an \emph{orthonormal sequence} if $v_k,v_l$ are orthogonal for $k\ne l$ ($\ip{v_k}{v_l} = 0$) and if $\|v_k\|^2 = \ip{v_k}{v_k} = 1$
\end{itemize}
\end{definition}

\begin{lemma}
\begin{itemize}
\item an orthogonal set is linearly-independent
\item an orthonormal sequence is linearly-independent
\end{itemize}
\end{lemma}
\end{frame}
