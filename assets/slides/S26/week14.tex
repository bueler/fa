\section{recall: Poisson problems and their finite element (FE) approximations}

\begin{frame}{recall: the Sobolev spaces $H^1(\Omega)$ and $H_0^1(\Omega)$}

\begin{definition}
    $$H^1(\Omega) = \left\{f\in L^2(\Omega) \,\Big|\, \begin{matrix} \text{the weak derivative } f' \text{ exists} \\ \text{and is in } L^2(\Omega)\end{matrix}\right\}$$
is a Hilbert space with inner product and norm:
	$$\ip{f}{g}_{H^1} = \int_\Omega f(x) g(x)\,dx + \int_\Omega \grad f(x)\cdot \grad g(x)\,dx, \qquad \|f\|_{H^1} = \sqrt{\ip{f}{f}_{H^1}}$$
\end{definition}

\begin{definition}
    $$H_0^1(\Omega) = \overline{C_c^1(\Omega)},$$
with closure in the $H^1$ norm
\end{definition}

\begin{itemize}
\item $H_0^1(\Omega)\subset H^1(\Omega)$ is the closed subspace of functions which have zero boundary values along $\partial \Omega$
\end{itemize}
\end{frame}


\begin{frame}{recall: the Poincar\'e-Friedrichs inequality}

\begin{theorem}[Poincar\'e-Friedrichs inequality]
Let $\Omega\subset \RR^d$ be an open, bounded domain with Lipschitz boundary.  There is $C>0$, depending only on $\Omega$, so that
	$$\|f\|_{L^2} \le C \|\grad f\|_{L^2} \quad \text{for all } f\in H_0^1(\Omega).$$
\end{theorem}

\begin{itemize}
\item a main consequence of this inequality is that the simpler formula
	$$|f|_{H^1} = \|\grad f\|_{L^2} = \left(\int_\Omega |\grad f|^2\right)^{1/2}$$
is \alert{equivalent to the full $H^1$ norm on $H_0^1(\Omega)$}
\item the full norm is $\|f\|_{H^1} = \left(\int_\Omega |f|^2 + \int_\Omega |\grad f|^2\right)^{1/2}$
\item \dots I still owe you a proof of this inequality
\end{itemize}
\end{frame}


\begin{frame}{recall: the weak form of example PDE problems}

\begin{itemize}
\item I'll use these two example elliptic PDEs to illustrate the FE theory
\end{itemize}

\begin{block}{Poisson equation weak form (homogeneous Dirichlet conditions)}
Let $V=H_0^1(\Omega)$.  Suppose $f\in L^2(\Omega)$ is given.  Find $u\in V$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \quad \forall v\in V$$
\end{block}

strong form Poisson: \, $\boxed{-\grad^2 u = f}$

\begin{block}{Helmholtz equation weak form (general Neumann conditions)}
Let $V=H(\Omega)$.  Suppose $f\in L^2(\Omega)$ and $g_N\in L^2(\partial\Omega)$ is given and $\gamma>0$.  Find $u\in V$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx + \gamma \int_\Omega u \cdot v\,dx = \int_\Omega f v\,dx + \int_{\partial \Omega} g_N v\,ds \quad \forall v\in V$$
\end{block}

strong form Helmholtz: \, $\boxed{-\grad^2 u + \gamma u = f}$
\end{frame}

\begin{frame}{recall: the basic idea of the finite element method}

\begin{block}{weak form (here for illustration: Poisson equation)}
Find $u\in V$ so that
    $$\int_\Omega \grad u \cdot \grad v\,dx = \int_\Omega f v\,dx \quad \forall v\in V$$
\end{block}

\begin{itemize}
\item FE method solves \alert{same weak form}, but over a finite-dim'l subspace $V_h$
\end{itemize}

\begin{block}{finite element method}
Find $u_h\in V_h$ so that
    $$\int_\Omega \grad u_h \cdot \grad v_h\,dx = \int_\Omega f v_h\,dx \quad \forall v_h\in V_h$$
\end{block}

\begin{itemize}
\item for Dirichlet conditions:
   $$V_h = \left\{f \in P_1\,:\,f\big|_{\partial\Omega} = 0\right\} \subset V=H_0^1(\Omega)$$
\item for Neumann conditions: \, $V_h = P_1 \subset V=H^1(\Omega)$
\end{itemize}
\end{frame}

\section{abstract bilinear form notation}

\begin{frame}{abstract bilinear notation}

\begin{itemize}
\item we write
    $$a:V\times V\to \RR$$
for the \alert{bilinear function} on the left side of the weak form
\begin{align*}
a(cu+dv,w) &= c\, a(u,w) + d\, a(v,w) \\
a(u,cv+dw) &= c\, a(u,v) + d\, a(u,w)
\end{align*}
\item thus $a(u,v) \in \RR$ if $u,v\in V$
\item for now we do \emph{not} assume it is symmetric (i.e.~$a(u,v)=a(v,u)$)
\end{itemize}

\begin{block}{examples: bilinear forms}
\begin{itemize}
\item Poisson equation: \, $\ds a(u,v) = \int_\Omega \grad u\cdot \grad v\,dx$
\item Helmholtz equation: \, $\ds a(u,v) = \int_\Omega \grad u\cdot \grad v\,dx + \gamma \int_\Omega u v\,dx$
\end{itemize}
\end{block}
\end{frame}


\begin{frame}{abstract linear notation}

\begin{itemize}
\item we write
    $$\ell:V \to \RR$$
for the \alert{linear function} on the right side of the weak form
\begin{align*}
\ell(cv+dw) &= c\, \ell(v) + d\, \ell(w)
\end{align*}
\item thus $\ell(v) \in \RR$ if $v\in V$
\item we will assume $\ell$ is continuous: $\ell \in V'$ is in the dual space
\end{itemize}

\begin{block}{examples: linear right-hand sides}
\begin{itemize}
\item Poisson equation w homogeneous Dirichlet: \, $\ds \ell(v) = \int_\Omega f v\,dx$
\item Helmholtz eqn w general Neumann: \, $\ds \ell(v) = \int_\Omega f v\,dx + \int_{\partial\Omega} g_N v\,ds$
\end{itemize}
\end{block}
\end{frame}


\begin{frame}{recall: minimization formulation leads to well-posedness}

\begin{itemize}
\item this notation was already used in a theorem
\end{itemize}

\begin{theorem}[well-posedness]
For $V$ a Hilbert space, suppose $a:V\times V\to \RR$ is a symmetric bilinear function.  Assume it is \emph{coercive}: there is $\alpha>0$ so that
	$$a(v,v) \ge \alpha \|v\|^2 \quad \forall v \in V.$$
Assume it is \emph{continuous}: there is $C\ge 0$ so that
	$$|a(u,v)| \le C \|u\|\|v\| \quad \forall u,v \in V.$$
Suppose $\ell\in V'$.  Then there exists $u\in V$ which is the unique minimizer of $J$:
	$$u \quad \stackrel{\min}{\leftarrow} \quad J(v) = \frac{1}{2} a(v,v) - \ell(v).$$
Furthermore $u\in V$ solves
	$$a(u,v) = \ell(v) \quad \forall v\in V.$$
\end{theorem}
\end{frame}


\begin{frame}{basic idea of the finite element method (again)}

\begin{itemize}
\item now we have very clean notation for the basic FE idea
\end{itemize}

\begin{block}{weak form}
Find $u\in V$ so that
    $$a(u,v) = \ell(v) \quad \forall v\in V$$
\end{block}

\begin{itemize}
\item FE method solves \alert{same weak form}, but over a finite-dim'l subspace $V_h$
\end{itemize}

\begin{block}{finite element method}
Find $u_h\in V_h$ so that
    $$a(u_h,v_h) = \ell(v_h) \quad \forall v_h\in V_h$$
\end{block}
\end{frame}


\section{C\'ea's lemma: the FE approximation is (essentially) the best}

\begin{frame}{conforming FE methods}

\begin{definition}
we say that an FE method is \emph{conforming} if
\begin{itemize}
\item \alert{$V_h\subset V$}, and
\item $a(\cdot,\cdot)$ and $\ell(\cdot)$ used in the FE method \alert{are the same} as those in the original PDE weak form
\end{itemize}
\end{definition}

\begin{itemize}
\item recall the problems together:
\begin{align*}
\text{find } u \in V \text{ so that} \quad a(u,v) &= \ell(v) \quad \forall v\in V \\
\text{find } u_h \in V_h \text{ so that} \quad a(u_h,v_h) &= \ell(v_h) \quad \forall v_h\in V_h
\end{align*}

\begin{definition}
the function $u-u_h$ is the FE \emph{error}
\end{definition}

\item in a conforming FE method we also know:
\begin{align*}
u - u_h &\in V \\
a(u,v_h) &= \ell(v_h) \quad \forall v_h\in V_h
\end{align*}
\end{itemize}
\end{frame}


\begin{frame}{Galerkin orthogonality}

\begin{itemize}
\item in a conforming FE method we have:
\begin{align*}
a(u,v_h) &= \ell(v_h) \quad \forall v\in V_h \\
a(u_h,v_h) &= \ell(v_h) \quad \forall v_h\in V_h
\end{align*}
\item subtract and use bilinear:
   $$a(u-u_h,v_h) = a(u,v_h) - a(u_h,v_h) = 0 \quad \forall v_h\in V_h$$
\end{itemize}

\begin{block}{Galerkin orthogonality}
   $$a(u-u_h,v_h) = 0 \quad \forall v_h\in V_h,$$
that is, \alert{the error $u-u_h$ is orthogonal to the entire FE space $V_h$}
\end{block}

picture:

\vspace{20mm}
\end{frame}


\begin{frame}{assumptions about the weak form PDE}

\begin{block}{assumptions for next slide}
\begin{itemize}
\item $\ell\in V'$ is linear and continuous
\item $a:V\times V\to \RR$ is a bilinear function
\item $a$ is coercive (elliptic): there is $\alpha>0$ so that
	$$a(v,v) \ge \alpha \|v\|^2 \qquad \forall v \in V$$
\item $a$ is continuous (bounded): there is $M\ge 0$ so that
	$$|a(u,v)| \le M \|u\|\|v\| \qquad \forall u,v \in V$$
\end{itemize}
\end{block}

\begin{itemize}
\item same assumptions as in the minimization approach
\item note $\alpha \le M$ \dots \emph{why?}
\item we'll check these assumptions in examples
\end{itemize}
\end{frame}


\begin{frame}{quasi-optimality, often called Cea's lemma}

\begin{lemma}[quasi-optimality]
Make the assumptions on the previous slide.  Then
	$$\|u-u_h\| \le \frac{M}{\alpha} \|u-v_h\| \quad \forall v_h \in V_h.$$
Thus, up to a factor of $M/\alpha$, \alert{$u_h$ is closer to the exact solution} than any other element of $V_h$.
\end{lemma}

{\small
\emph{Proof.}  Let $v_h\in V_h$ be arbitrary.  By coercivity, bilinearity, Galerkin orthogonality, and continuity:
\begin{align*}
\alpha \|u-u_h\|^2 &\le a(u-u_h,u-u_h) \\
&= a(u-u_h,u-v_h+v_h-u_h) = a(u-u_h,u-v_h) + a(u-u_h,v_h-u_h) \\
&= a(u-u_h,u-v_h) + 0 \\
&= M \|u-u_h\| \|u-v_h\|
\end{align*}
Dividing by $\|u-u_h\|$ gives $\alpha \|u-u_h\| \le M \|u-v_h\|$, as claimed. \hfill \qed
}
\end{frame}


\begin{frame}{example: homogeneous Dirichlet problem for Poisson equation}

\begin{itemize}
\item here $V=H_0^1(\Omega)$, with norm $\|v\|_{H^1} = \left(\|v\|_{L^2} + \|\grad v\|_{L^2}\right)^{1/2}$
\item $\ell$ is clearly linear, and bounded via CS, and $a$ is clearly bilinear
\item coercivity is from Poincar\'e-Friedrichs: $\|\phi\|_{L^2} \le C \|\grad\phi\|_{L^2}$
\item specifically, we may take $\alpha=\frac{1}{2} \min\{1,1/C^2\}$:
\begin{align*}
a(v,v) &= \int_\Omega |\grad v|^2\,dx = \|\grad v\|_{L^2}^2 \ge \frac{1}{2} \|\grad v\|_{L^2}^2 + \frac{1}{2} \|\grad v\|_{L^2}^2 \\
&\ge \frac{1}{2} \|\grad v\|_{L^2}^2 + \frac{1}{2C^2} \|v\|_{L^2}^2 \ge \alpha \|v\|_{H^1}^2
\end{align*}
\item $a$ is continuous via Cauchy-Schwarz, with $M=1$:
	$$|a(u,v)| \le \int_\Omega |\grad u||\grad v|\,dx \le \|\grad u\|_{L^2} \|\grad v\|_{L^2} \le \|u\|_{H^1} \|v\|_{H^1}$$
\end{itemize}

\vspace{-2mm}
\begin{corollary}[quasi-optimality in $H^1$ norm]
	$$\|u-u_h\|_{H^1} \le \frac{1}{\alpha} \|u-v_h\|_{H^1} \quad \forall v_h\in V_h$$
\end{corollary}
\end{frame}

\begin{frame}{example: homogeneous Dirichlet problem for Poisson equation 2}

\begin{itemize}
\item we can do better in this case!
\item switch the norm on $V=H_0^1(\Omega)$ to \quad \alert{$\vertiii{v}=\|\grad v\|_{L^2}$}
    \begin{itemize}
    \item[$\circ$] Poincar\'e-Friedrichs says this is a norm
    \item[$\circ$] called the \emph{energy norm}, by poorly-motivated traditional reasons
    \end{itemize}
\item now $a$ is coercive with $\alpha=1$,
    $$a(v,v) = \int_\Omega |\grad v|^2\,dx = \vertiii{v}^2 \ge 1\,\vertiii{v}^2,$$
\item it is easy to see that $a$ is continuous with $M=1$
\item the quasi-optimality lemma becomes \dots
\end{itemize}

\begin{corollary}[optimality in energy norm]
	$$\vertiii{u-u_h} \le \vertiii{u-v_h} \quad \forall v_h\in V_h$$
\end{corollary}

\begin{itemize}
\item this is \alert{actual optimality} in this norm
\end{itemize}
\end{frame}


\begin{frame}{example: general Neumann problem for Helmholtz equation}

\begin{itemize}
\item recall Helmholtz: \, $\ds a(u,v) = \int_\Omega \grad u\cdot \grad v\,dx + \gamma \int_\Omega u v\,dx$
\item here $V=H^1(\Omega)$, with usual norm $\|v\|_{H^1} = \left(\|v\|_{L^2} + \|\grad v\|_{L^2}\right)^{1/2}$
\item coercivity does not need PF
\item in fact, $\alpha=\min\{1,\gamma\}$:
\begin{align*}
a(v,v) &= \int_\Omega |\grad v|^2\,dx + \gamma \int_\Omega |v|^2\,dx \\
     &\ge \alpha \left(\int_\Omega |\grad v|^2\,dx + \int_\Omega |v|^2\,dx\right) = \alpha \|v\|_{H^1}^2
\end{align*}
\item $a$ is continuous via Cauchy-Schwarz, with $M=\max\{1,\gamma\}$:
	$$|a(u,v)| \le \int_\Omega |\grad u||\grad v|\,dx + \gamma \int_\Omega |u||v|\,dx \le M \|u\|_{H^1} \|v\|_{H^1}$$
\end{itemize}

\begin{corollary}[quasi-optimality in $H^1$ norm]
	$$\|u-u_h\|_{H^1} \le \frac{\max\{1,\gamma\}}{\min\{1,\gamma\}} \|u-v_h\|_{H^1} \quad \forall v_h\in V_h$$
\end{corollary}
\end{frame}

\begin{frame}{how will quasi-optimality show that the FE method converges?}

\begin{itemize}
\item let us assume about the original, continuous PDE problem:
    \begin{enumerate}
    \item the weak form is well-posed and determines a unique solution $u$
    \item \alert{the exact solution $u$ is a smooth function on $\Omega$}
    \end{enumerate}
\item in the problems so far we know $u\in H^1(\Omega)$, but it is often true that $u\in H^2(\Omega)$ or $u\in C^2(\Omega)$ or even $u \in C^\infty(\Omega)$
    \begin{itemize}
    \item[$\circ$] a \emph{regularity} theorem would supply such knowledge
    \end{itemize}
\item now we make a definition related to the FE problem
\end{itemize}

\begin{definition}
on a given mesh $\mathcal{T}$ over $\Omega$, with a given value $h>0$, there is an interpolation map
	$$\II_h : C(\overline{\Omega}) \to P_1$$
which takes the nodal values of the input $w\in C(\overline{\Omega})$ and constructs the element of $P_1$ with those nodal values
\end{definition}
\end{frame}


\begin{frame}{how will quasi-optimality show that the FE method converges?}

\begin{itemize}
\item we need an \emph{interpolation theorem} to say how close $\II_h(w)$ is to $w$

\begin{theorem}[generic interpolation theorem result]
    $$\alert{\|w-\II_h(w)\| = O(h^k)}$$
\end{theorem}

    \begin{itemize}
    \item[$\circ$] actual theorems later in these slides, for 1D intervals and 2D triangles
    \item[$\circ$] such theorems provide a rate $O(h^k)$ for interpolation accuracy, given $w\in H^\ell(\Omega)$, for $\Omega\subset \RR^d$, with relationships between $k,\ell,d$
    \end{itemize}
\item then we \alert{substitute $v_h = \II_h(u)$ into quasi-optimality} and use the rate:
	$$\|u-u_h\| \le \frac{M}{\alpha} \|u-\II_h(u)\| = O(h^k)$$
\item thus $u\to u_h$ as $h\to 0$, so the method \alert{converges}
\end{itemize}
\end{frame}


\begin{frame}{before moving on \dots Cea's actual lemma is better!}

\begin{lemma}[Cea's lemma \dots from his 1964 PhD thesis]
Make the same assumptions about our weak form, and also assume that the bilinear form is \alert{symmetric} $a(u,v)=a(v,u)$.  Then
	$$\|u-u_h\| \le \sqrt{\frac{M}{\alpha}} \|u-v_h\|$$
for all $v_h\in V_h$.  (The constant $\sqrt{M/\alpha}$ is smaller than $M/\alpha$, thus \alert{improved} relative to the original lemma.)
\end{lemma}

{\footnotesize
\emph{Proof.}  Let $v_h\in V_h$ be arbitrary.  By bilinearity, then Galerkin orthogonality, and then $a(u_h-v_h,u_h-v_h)\ge 0$ from coercivity:
\begin{align*}
a(u-v_h,u-v_h) &= a(u-u_h+u_h-v_h,u-u_h+u_h-v_h) \\
&= a(u-u_h,u-u_h) + 2 a(u-u_h,u_h-v_h) + a(u_h-v_h,u_h-v_h) \\
&= a(u-u_h,u-u_h) + 0 + a(u_h-v_h,u_h-v_h) \\
&\ge a(u-u_h,u-u_h)
\end{align*}
Thus by coercivity (again), and continuity, we get the claim:
\begin{align*}
\alpha \|u-u_h\|^2 &\le a(u-u_h,u-u_h) \le a(u-v_h,u-v_h) \le M \|u-v_h\|^2 \quad \qed
\end{align*}
}
\end{frame}

\begin{frame}{Lagrange interpolation}

\begin{itemize}
\item methods for polynomial interpolation in $\RR^1$ are old (Vandermonde 1771, Waring 1779, Lagrange 1795)

\begin{block}{polynomial interpolation in 1D (Lagrange's formula)}
Given $n+1$ points $(x_i,y_i)$ for $i=0,1,\dots,n$, with distinct $x_i$.  There exists a unique polynomial $p(x)$ of degree $n$ such that \alert{$p(x_i)=y_i$} for all $i$, and in fact
	$$p(x) = \sum_{i=0}^n y_i L_i(x)$$
where $\ds L_i(x) = \prod_{\begin{smallmatrix} k = 0 \\ k\ne i \end{smallmatrix}}^n \frac{x-x_k}{x_i-x_k}$ is a polynomial of degree $n$ satisfying $\ds \alert{L_i(x_j) = \delta_{ij}}$.
\end{block}

\begin{theorem}[polynomial interpolation remainder]
Suppose $f\in C^{n+1}([a,b])$ and $x_i$, $i=0,1,\dots,n$, are distinct points in $[a,b]$.  Let $p(x)$ be the unique polynomial $p(x)$ of degree $n$ such that $p(x_i)=f(x_i)$.  Then
	$$f(x) = p(x) + R(x)$$
where
    $$\alert{R(x) = \frac{f^{(n+1)}(\xi)}{(n+1)!} (x-x_0)(x-x_1)\dots(x-x_n)}$$
and \alert{$\xi=\xi(x)$} is some point in $[a,b]$.
\end{theorem}

\begin{itemize}
\item obviously: \, $R(x)=f(x)-p(x)$
\end{itemize}

\begin{theorem}[polynomial interpolation remainder]
Let $p(x)$ be the unique polynomial of degree $n$ such that $p(x_i)=f(x_i)$.  Then $f(x) = p(x) + R(x)$ where
    $$R(x) = \frac{f^{(n+1)}(\xi)}{(n+1)!} (x-x_0)(x-x_1)\dots(x-x_n)$$
for some point $\xi=\xi(x) \in [a,b]$.
\end{theorem}

\begin{theorem}[Taylor's theorem with remainder]
Let $p(x)$ be the Taylor polynomial of degree $n$ at basepoint $x_0$.  Then $f(x) = p(x) + R(x)$ where
    $$R(x) = \frac{f^{(n+1)}(c)}{(n+1)!} (x-x_0)^{n+1}$$
for some point $c=c(x)$ between $x_0$ and $x$.
\end{theorem}
\end{frame}


\begin{frame}{proof in the $n=1$ (linear) case}

\begin{theorem}[linear interpolation remainder]
Suppose $f\in C^2([a,b])$.  For nodes $a=x_0<x_1=b$, let $p(x)$ be the unique linear polynomial such that $p(a)=f(a)$ and $p(b)=f(b)$.  Then for each $x\in [a,b]$ we have $f(x) = p(x) + R(x)$ where
    $$R(x) = \frac{f^{''}(\xi)}{2} (x-a)(x-b),$$
for some point $\xi \in (a,b)$ which depends on $x$.
\end{theorem}

\medskip
{\footnotesize
Proof.  Interpolation is exact at ends ($R(a)=R(b)=0$), so fix $x\in (a,b)$.  For $t\in [a,b]$ define
	$$g(t) = f(t) - p(t) - \frac{f(x)-p(x)}{(x-a)(x-b)} (t-a)(t-b).$$
Note that $g$ is $C^2$, and that it has three zeros: $g(a)=g(x)=g(b)=0$.  By Rolle's theorem there are points $a<c<x<d<b$ so that $g'(c)=0$ and $g'(d)=0$.  Applying Rolle's theorem again there is $c<\xi<d$ so that $g''(\xi)=0$.  However,
	$$0=g''(\xi)=f''(\xi) - 0 - \frac{f(x)-p(x)}{(x-a)(x-b)} 2,$$
as claimed. \hfill \qed
}
\end{frame}


\begin{frame}{piecewise-linear interpolation in $\RR^1$}

\begin{itemize}
\item we are primarily interested in the \alert{piecewise-linear} FE space $P_1$
\item suppose $\Omega=(a,b) \subset \RR^1$ and $f\in C^2(\overline\Omega)$
\item consider a mesh $a=x_0<x_1<\dots<x_{n-1}<x_n=b$; \quad picture:

\vspace{20mm}
\item let $p(x)$ be the continuous, piecewise-linear interpolant of $f(x)$
    \begin{itemize}
    \item[$\circ$] on $[x_{i-1},x_i]$: \quad $\ds \alert{p(x) = f(x_{i-1}) + \frac{f(x_i) - f(x_{i-1})}{x_i - x_{i-1}} (x-x_{i-1})}$
    \end{itemize}
\item the previous theorem gives a bound on $[x_{i-1},x_i]$:
	$$|f(x) - p(x)| \le \frac{M}{2} |(x-x_{i-1})(x-x_i)|$$
where $\ds M=\max_{\xi\in [x_{i-1},x_i]} |f''(\xi)|$
\end{itemize}
\end{frame}


\begin{frame}{piecewise-linear interpolation error bound}

\begin{itemize}
\item let $h = \max_i |x_i-x_{i-1}|$, and compute:
    $$\max_{x \in [x_{i-1},x_i]} |(x-x_{i-1})(x-x_i)| = \frac{1}{4} |x_{i+1}-x_i|^2 \le \frac{1}{4} h^2$$
\item thus on every subinterval ($=$ cell or element) $x\in[x_{i-1},x_i]$,
	$$\alert{|f(x) - p(x)| \le \frac{\|f''\|_\infty}{8} h^2}$$
\end{itemize}

\begin{theorem}[uniform interpolation bound in $\RR^1$]
Let $f\in C^2([a,b])$.  For a mesh $\{x_i\}$ on $[a,b]$ with $h=\max |x_i-x_{i-1}|$, consider the piecewise-linear interpolant $p=\II_h f \in P_1$.  Then
	$$\|f-\II_h f\|_\infty \le \frac{\|f''\|_\infty}{8} h^2.$$
\end{theorem}
\end{frame}


\begin{frame}{how about the $H^1$ norm?}

\begin{itemize}
\item we also want an error bound in the $H^1$ norm, so we need a lemma
\end{itemize}

\begin{lemma}
Suppose $f\in C^2([a,b])$ and suppose $[x_{i-1},x_i] \subset [a,b]$.  Let $p(x)$ be the unique linear polynomial such that $p(x_j)=f(x_j)$.  Then
    $$\int_{x_{i-1}}^{x_i} |f'(x)-p'(x)|^2\,dx \le h^3 \|f''\|_\infty^2$$
where $h=\max_i |x_i-x_{i-1}|$.
\end{lemma}

\medskip
{\small
Proof. Note $p'(x)$ is constant in $[x_{i-1},x_i]$.  By the mean value theorem there is $c\in (x_{i-1},x_i)$ so that $f'(c)$ has this value.  By FTC and triangle inequality:
\begin{align*}
\int_{x_{i-1}}^{x_i} |f'(x)-p'(x)|^2\,dx &= \int_{x_{i-1}}^{x_i} |f'(x)-f'(c)|^2\,dx = \int_{x_{i-1}}^{x_i} \left|\int_x^c f''(t)\,dt\right|^2\,dx \\
     &\le \int_{x_{i-1}}^{x_i} \left(\int_{x_{i-1}}^{x_i} |f''(t)|\,dt\right)^2\,dx \le (x_i - x_{i-1})^3 \|f''\|_\infty^2. \hspace{5mm} \qed
\end{align*}
}
\end{frame}


\begin{frame}{decay rate of $P_1$ interpolation error in $H^1$ and $L^2$}

\begin{theorem}[interpolation error rate for $P_1$ \alert{in $\RR^1$}]
Let $\Omega=(a,b)$ and $f\in C^2(\overline{\Omega})$.  Consider a sequence of meshes $\{x_i\}$ on $\Omega$ with $h=\max_i |x_i-x_{i-1}|$.  Suppose the meshes are quasi-uniform in the sense that there is $c>0$ so that $c h \le |x_i-x_{i-1}| \le h$ for all $i$.  The piecewise-linear interpolants\, $\II_h f \in P_1$ satisfy
\begin{align*}
\|f-\II_h f\|_{L^2} &= O(h^2) \|f''\|_\infty \\
\|f-\II_h f\|_{H^1} &= O(h^1) \|f''\|_\infty
\end{align*}
as $h\to 0$.
\end{theorem}

\medskip
{\small
Proof.  Let $p=\II_h f$.  From the uniform bound,
\begin{align*}
\|f-p\|_{L^2}^2 &= \int_a^b |f(x)-p(x)|^2\,dx \le \int_a^b \|f-p\|_\infty^2\,dx \\
  &\le (b-a) \left(\frac{\|f''\|_\infty}{8} h^2\right)^2 = C_1 h^4 \|f''\|_\infty^2
\end{align*}
Thus $\|f-p\|_{L^2} \le \sqrt{C_1} h^2 \|f''\|_\infty$, which proves the first big-O claim.
}
\end{frame}


\begin{frame}{decay rate (proof continued)}

{\small
Proof cont.  Now using the lemma,
\begin{align*}
\|f'-p'\|_{L^2}^2 &= \sum_{i=1}^n \int_{x_{i-1}}^{x_i} |f'(x)-p'(x)|^2\,dx \le n h^3 \|f''\|_\infty^2
\end{align*}
Note that by quasi-uniformity of the mesh, $nh\le (b-a)/c$.  Thus
	$$\|f'-p'\|_{L^2}^2 \le \frac{b-a}{c} h^2 \|f''\|_\infty^2$$
Combining these results,
\begin{align*}
\|f-p\|_{H^1}^2 &= \|f-p\|_{L^2}^2 + \|f'-p'\|_{L^2}^2 \le C_1 h^4 \|f''\|_\infty^2 +  \frac{b-a}{c} h^2 \|f''\|_\infty^2 \\
  &\le (C_1 h^4 + \frac{b-a}{c} h^2) \|f''\|_\infty^2 \le C_2 h^2 \|f''\|_\infty^2.
\end{align*}
(Restricting to $0<h\le 1$, take $C_2=C_1+(b-a)/c$.)  Thus $\|f-p\|_{H^1} \le \sqrt{C_2} h \|f''\|_\infty$ as claimed. \hfill \qed
}
\end{frame}
