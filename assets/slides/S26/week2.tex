\begin{frame}{Lipschitz kernels}

\begin{itemize}
\item regarding what conditions on $k(x,t)$ imply $L\,:\,C([0,1])\to C([0,1])$, here is a sufficient condition
\end{itemize}

\begin{definition}
a function $k:[0,1]\times [0,1] \to \RR$ is \emph{Lipschitz in $x$} if there is $C\ge 0$ so that for all $x,y \in [0,1]$ and $t\in[0,1]$, \quad $|k(x,t)-k(y,t)|\le C |x-y|$
\end{definition}

\begin{lemma}
if $k(x,t)$ is Lipschitz in $x$, and if $(L\phi)(x) = \int_0^1 k(x,t) \phi(t)\,dt$ for $\phi \in C([0,1])$, then $L\phi \in C([0,1])$
\end{lemma}

\begin{frame}{Lipschitz kernels}

\begin{lemma}
if $k(x,t)$ is Lipschitz in $x$, and if $(L\phi)(x) = \int_0^1 k(x,t) \phi(t)\,dt$ for $\phi \in C([0,1])$, then $L\phi \in C([0,1])$
\end{lemma}

{\small
\noindent \emph{Proof.}  Let $\phi\in C([0,1])$, and consider the function $L\phi$ at a point $x\in[0,1]$.  Suppose $\eps>0$.  Let $\delta=\eps/(C\|\phi\|_\infty)$.  If $y\in[0,1]$ and $|x-y|<\delta$ then by the triangle inequality for integrals, and the definition of Lipschitz,
\begin{align*}
\left|(L\phi)(x) - (L\phi)(y)\right| &= \left|\int_0^1 k(x,t) \phi(t)\,dt - \int_0^1 k(y,t) \phi(t)\,dt\right| \\
  &\le \int_0^1 \left|k(x,t) - k(y,t)\right| |\phi(t)| \,dt \le \int_0^1 C |x-y|\, |\phi(t)| \,dt \\
  &\le \int_0^1 C|x-y|\, \|\phi\|_\infty \,dt = C \|\phi\|_\infty |x-y| < C \|\phi\|_\infty \delta = \eps \qquad \qed
\end{align*}
}
\end{frame}

\begin{frame}{compact sets}

\begin{itemize}
\item now Chapter 2 of \textbook
\item we assume $(V,\|\cdot\|)$ is a normed vector space
\item recall: a subset $S\subset V$ is \emph{open} if $\forall$ $x\in S$ $\exists$ $r>0$ so that $B_r(x)\subset S$

     \begin{itemize}
     \item[$\circ$] $B_r(x)=\{y\in V\,:\,\|y-x\| < r\}$ is an \emph{open ball}
     \end{itemize}
\end{itemize}

\begin{definition}
$\mathcal{C}=\{S_\lambda\}_{\lambda\in I}$ is an \emph{open cover of } $E\subset V$ if $S_\lambda\subset V$ is open for each $\lambda \in I$ and $E \subset \bigcup_{\lambda\in I} S_\lambda$
\end{definition}

\begin{definition}
\begin{itemize}
\item a subset $E\subset V$ is \emph{compact} if every open cover of $E$ has a finite subcover
\item a subset $E\subset V$ is \emph{sequentially compact} if every sequence in $E$ contains a convergent subsequence
\end{itemize}
\end{definition}

\begin{itemize}
\item fact: a finite set is compact
\end{itemize}

\begin{block}{Theorem 2.4}
$E\subset V$ is compact if and only if it is sequentially compact
\end{block}

\end{frame}


\begin{frame}{the unit ball in $V=C([0,1])$ is not compact}

\begin{block}{Theorem 2.3}
if $E\subset V$ is compact then it is closed
\end{block}

\begin{theorem}
the closed unit ball $B_1(0)=\{y\in V=C([0,1])\,:\,\|y\|_\infty \le 1\}$ is not compact
\end{theorem}

{\footnotesize
\noindent \emph{Proof.}  The hat functions $\psi_n(x)$, $n\in \NN$, drawn below, are continuous, and each has norm $\|\psi_n\|_\infty = 1$, thus $\psi_n \in B_1(0)$.  Also, for any $n\ne m$, $\|\psi_n - \psi_m\|_\infty = 1$.  Any subsequence $(\psi_{n_k})$ of the sequence $(\psi_n)$ would not converge because again $\|\psi_{n_j} - \psi_{n_k}\|_\infty = 1$ for $j\ne k$.  Thus $B_1(0)$ is not compact.  \hfill \qed
}

\vspace{20mm}

\begin{itemize}
\item \emph{Corollary.} if $E \subset V=C([0,1])$ contains a ball of positive radius $B_r(x) \subset E$, no matter how small $r>0$ is, then $E$ is not compact
\end{itemize}
\end{frame}


\begin{frame}{why would we want sets to be compact?}

\begin{block}{Extreme Value Theorem (Weierstrauss)}
if $E\subset V$ is compact and $f:E\to\RR$ is continuous then there exists a maximum and a minimum of $f$
\end{block}

\begin{itemize}
\item here $(V,\|\cdot\|)$ is any normed vector space
\item the conclusion is that there are $c,C\in E$ so that
	$$f(c) \le f(x) \le f(C)$$
for all $x\in E$
\item slightly more generally, one can consider continuous $f:M\to \RR$ where $(M,d)$ is any compact metric space
\item compact subsets of $V=C([0,1])$ are a major focus of Chapter 2
\end{itemize}

\end{frame}


\section{bounded and compact operators}

\begin{frame}{bounded and compact operators}

\begin{itemize}
\item assume $(V,\|\cdot\|)$ is a normed vector space
\item the closed unit ball is $B =\{y\in V\,:\,\|y\| \le 1\}$
\end{itemize}

\begin{definition}
\begin{itemize}
\item a linear operator $L:V \to V$ is \emph{bounded} if there is a constant $C\ge 0$ so that
    $$\|Lv\|\le C \|v\| \quad \text{ for all } v\in V$$
\item a linear operator $L:V \to V$ is \emph{compact} if the image of the closed unit ball under $L$, that is, the set $LB$, is compact
\end{itemize}
\end{definition}

\begin{itemize}
\item we will see that ``bounded'' is the same as ``continuous'' for \emph{linear} operators
\end{itemize}

\end{frame}


\begin{frame}{bounded integral operators on $V=C([0,1])$}

\begin{theorem}
if $k(x,t)$ is integrable, and if
    $$\sup_{x \in [0,1]} \int_0^1 |k(x,t)|\,dt<\infty,$$
then the operator $L:V\to V$, \, $(L\phi)(x) = \int_0^1 k(x,t) \phi(t)\,dt$, \, is bounded
\end{theorem}

{\footnotesize
\noindent \emph{Proof.}  Suppose $\phi \in V$.  Then by the triangle inequality,
\begin{align*}
\|L\phi\|_\infty &= \sup_{x \in [0,1]} |(L\phi)(x)| = \sup_{x \in [0,1]} \left|\int_0^1 k(x,t) \phi(t)\,dt \right| \\
   &= \sup_{x \in [0,1]} \int_0^1 |k(x,t)| |\phi(t)|\,dt \le \sup_{x \in [0,1]} \int_0^1 |k(x,t)| \|\phi\|_\infty \,dt \\
   &= \left(\sup_{x \in [0,1]} \int_0^1 |k(x,t)| \,dt\right)\|\phi\|_\infty
\end{align*}
Thus $\|L\phi\|_\infty \le C \|\phi\|_\infty$ where $C=\sup_{x \in [0,1]} \int_0^1 |k(x,t)| \,dt$. \hfill \qed
}

\end{frame}


\begin{frame}{new idea: equicontinuous \underline{sets} of functions}

\begin{definition}
a set $E \subset C([a,b])$ is \emph{equicontinuous at } $x\in[a,b]$ if for each $\eps>0$ there is $\delta >0$ so that $y\in[a,b]$ and $|x-y|<\delta$ implies $|f(x)-f(y)|<\eps$ for all $f\in E$
\end{definition}

\begin{itemize}
\item this is about quantifiers!
\item $f$ continuous at $x \in [a,b]$:
    $$\forall \eps>0 \; \exists\, \delta > 0 \; \forall y \in [a,b] \quad |x-y|<\delta \implies |f(x)-f(y)|<\eps$$
\item $E$ equicontinuous at $x \in [a,b]$:
    $$\forall \eps>0 \; \alert{\forall f\in E} \; \exists\, \delta > 0 \; \forall y \in [a,b] \quad |x-y|<\delta \implies |f(x)-f(y)|<\eps$$
\end{itemize}

\begin{definition}
a set $E \subset C([a,b])$ is \emph{equicontinuous} if it is equicontinous at each $x\in [a,b]$
\end{definition}

\begin{itemize}
\item more on equicontinuity in Chapter 2 of \textbook
\end{itemize}
\end{frame}


\begin{frame}{new idea: some integral operators are compact}

\begin{block}{Ascoli-Arzel\`a Theorem}
a set $E\subset C([a,b])$ is compact if and only if it is closed, bounded, and equicontinuous
\end{block}

\begin{itemize}
\item now we can return to the BVP!
\end{itemize}

\begin{block}{Theorem}
Consider $B_R(0)\subset V=C([0,1])$.  For $\phi \in B_R(0)$ the solution of
    $$-u'' = \phi, \quad u(0) = u(1) = 0$$
is \, $u(x) = (L\phi)(x) = \int_0^1 k(x,t) \phi(t)\,dt$ \, where $k(x,t) = \min\{x,t\} - xt$.  Then $u\in E$ where $E= LB_R(0)$ is a compact subset of $V$.
\end{block}

{\footnotesize
\noindent \emph{Proof.}  Show $E$ is closed and bounded.  Show $E$ equicontinuous by using the fact that $k(x,t)$ is Lipschitz in $x$. \hfill \qed
}
\end{frame}
