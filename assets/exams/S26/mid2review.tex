\noindent \textbf{Definitions}.  Be able to state the precise definition.  Be able to prove ideas which follow immediately from the definitions.
\begin{itemize}
\item abstract measure space $(X,\cR,\mu)$
\item almost everywhere
\item $\int_A f\,d\mu$, the (abstract) Lebesgue integral of a real-valued or complex-valued function $f$ over $A$ with respect to the measure $\mu$
\item for $1\le p < \infty$, the vector space $L^p(X,\mu)$ is the set of equivalence classes of functions, up to almost everywhere, such that $\int_X |f|^p\,d\mu<\infty$, with norm $\|f\|_p = \left(\int_X |f|^p\,d\mu\right)^{1/p}$
\item essentially-bounded measurable function
\item the vector space $L^\infty(X,\mu)$ is the set of equivalence classes of essentially-bounded functions, up to almost everywhere, with the essential bound as the norm $\|f\|_\infty$
\item complete orthonormal (ON) sequence in an inner product space
\item Fourier coefficients and Fourier series with respect to an ON sequence
\item absolute continuity (classical definition for functions on $\RR^1$), and $AC([a,b])$
\item $\operatorname{supp}(f)$, for a real-valued or complex-valued function on an open set $\Omega\subset \RR^d$
\item $C_c(\Omega)$, $C_c^k(\Omega)$, $C_c^\infty(\Omega)$, on open sets $\Omega\subset \RR^d$
\item $D^\alpha f$, multi-index notation for partial derivatives on $\RR^d$
\item $L_{loc}^1(\Omega)$, on open sets $\Omega\subset \RR^d$
\item weak derivatives of order $\alpha$, for functions on open sets $\Omega\subset \RR^d$
\item bounded linear operator, between normed vector spaces
\item $\mathcal{B}(X,Y)$, for normed vector spaces $(X,\|\cdot\|_X)$ and $(Y,\|\cdot\|_Y)$
\item operator norm, making $\mathcal{B}(X,Y)$ into a normed vector space
\item dual space $X'$, for a normed vector space $(X,\|\cdot\|_X)$
\item Laplacian $\nabla^2 u$, on differentiable functions $u$ on $\Omega \subset \RR^d$
\item harmonic function
\item $H^k(\Omega)$, for $k=0,1,2,\dots$ and an open set $\Omega\subset \RR^d$
\item $H^k(S^1)$ (there are two equivalent definitions; know both)
\item $H_0^1(\Omega)$, for an open set $\Omega\subset \RR^d$
\end{itemize}

\noindent \textbf{Theorems and Lemmas}.  Know the precise statement of the theorem.  Be able to prove if so indicated.
\newcommand{\bap}{\, {\small\textbf{[be able to prove]}}}
\begin{itemize}
\item Lebesque's monotone convergence theorem (Theorem 3.13)
\item Fatou's lemma (Theorem 3.14)
\item Lebesque's dominated convergence theorem (Theorem 3.15)
\item for $1\le p \le \infty$, $L^p(X,\mu)$ is a complete normed linear space (Theorems 3.17, 3.19, 3.21, and text on page 61 for $L^\infty$)
\item H\"older's inequality (Theorem 3.18)
\item normed vector space is complete$\iff$(absolute conv.$\implies$conv.) (Theorem 3.20)
\item step functions are dense in $L^p(X,\mu)$ (see \href{https://bueler.github.io/fa/corrections.html}{Corrections} tab of the public website)
\item Fourier coefficients are $c_k=\ip{f}{f_k}$ if $\{f_k\}$ is a complete ON sequence (Theorem 4.1)
\item bounded$\iff$continuous$\iff$cont.~at 0 (Theorems 5.1 \& 5.2) \bap
\item operator norm is a norm on $\mathcal{B}(X,Y)$ (Theorem 5.3) \bap
\item $\mathcal{B}(X,Y)$ is a Banach space if $Y$ is a Banach space (Theorem 5.4)
\item corollary: dual spaces are Banach spaces \bap
\item Hilbert projection theorem: if $(X,\ip{\cdot}{\cdot})$ is a Hilbert space and $C\subset X$ is a closed and convex subset then for every $x\in X$ there exists a unique $y\in C$ that minimizes $f(w)=\|x-w\|$ over $w\in C$
\item Hilbert orthogonal decomposition theorem: if $(X,\ip{\cdot}{\cdot})$ is a Hilbert space and $M\subset X$ is a closed subspace then $X = M\oplus M^{\perp}$
\item Riesz-Fr\'echet representation theorem: if $(X,\ip{\cdot}{\cdot})$ is a Hilbert space and $\ell\in X'$ then there exists a unique $x_\ell\in X$ so that $\ell(x) = \ip{x}{x_\ell}$ for all $x\in X$
\item fundamental theorem of calculus: if $f\in L^1(a,b)$ then $F(x)=\int_a^x f(t)\,dx$ is in $AC([a,b])$ and $F'(x)=f(x)$ a.e., and if $f \in AC([a,b])$ then $f'\in L^1(a,b)$ and $f(x) = f(a) + \int_a^x f'(t)\,dt$ (week 10\&11 slides)
\item divergence theorem: if $\Omega\subset \RR^d$ is bounded and $\partial\Omega$ is $C^1$, and if $\bV\in C^1(\overline\Omega)$, then $\int_\Omega \Div\bV\,dm = \int_{\partial\Omega} \bV\cdot \bn\,ds$ (week 10\&11 slides)
\item integration-by-parts on $(a,b) \subset \RR^1$ (week 10\&11 slides) \bap
\item product rule 1 and 2 (week 10\&11 slides)
\item integration-by-parts 1 and 2 (week 10\&11 slides) \bap
%\item Poincar\'e-Friedrichs inequality (week 12 slides) 
\end{itemize}
