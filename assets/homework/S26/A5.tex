\noindent \textsc{Do the following additional problems.}

\prob{P7}  \emph{The following problem asks for the heart of the correct proof of Theorem 3.22, in the $p=1$ and $\RR^1$ case.  \href{https://bueler.github.io/fa/corrections.html}{The book's definition of step functions is wrong}, but the Theorem is true.}

\medskip \noindent A function $h:X \to \RR$ is a \emph{step function} if it is a finite linear combination of characteristic functions of intervals; there are coefficients $c_k \in \RR$ and intervals $I_k\subset \RR$ so that
   $$h(x) = \sum_{k=1}^n c_k \one_{I_k}(x).$$

\medskip \noindent Suppose $E\subset [0,1]$ is measurable.  Show that the indicator function $f(x)=\one_E(x)$ can be arbitrarily approximated in $L^1$ norm by a step function.  That is, show that for all $\eps>0$ there exists a step function $g(x)$ so that
	$$\|\one_E-g\|_1 = \int_{[0,1]} |\one_E(x) - g(x)|\,dm < \eps.$$
(\emph{Hint. Return to the definition of $\cM_{\cF}$.  Use it.  The function $g$ is also an indicator.})


\prob{P8}  {\color{red} \emph{(Simplified.)}}  Prove that the triple $(X,\cR,\mu)$ given in Example 6 at the beginning of Section 3.6 (page 58) is, in fact, a measure space.


\prob{P9} {\color{red} \emph{(Simplified: part (b) removed.)}}   Suppose that $(X,\cR,\mu)$ is a measure space.  We say that two measurable functions $f,g:X\to\RR$ are \emph{equal almost everywhere} if the set of points where they differ has $\mu$-measure zero.  Prove that this is an equivalence relation on the set of such measurable functions.


\prob{P10}  Suppose that $(X,\cR,\mu)$ is a \emph{finite} measure space, that is, suppose that $X\in\cR$ and $\mu(X)<\infty$.  For $f:X\to\CC$ measurable (with respect to $\cR$) and $1\le p<\infty$, define $L^p(X,\mu)$ and $\|f\|_p$ as usual; see page 59.  Consider two different powers, $1\le p<q<\infty$.  Show that $L^q(X,\mu) \subset L^p(X,\mu)$.  Give an example of a function on $(X,\mu)=([0,1],m)$, which will depend on $p$ and $q$, for which this is a proper inclusion.


\prob{P11}  \emph{The requested proofs will help you appreciate H\"older's and Minkowski's inequalities.  These results are used in Theorems 3.17, 3.18, and 3.19.  For part} \textbf{(a)}\emph{, start with the triangle inequality on $\RR$, and also $|u|+|v| \le 2\max(|u|,|v|)$.  A hint on part} \textbf{(b)} \emph{is on page 60.}

\epart{a} Suppose $1\le p<\infty$ and let $u,v\in\RR$.  Show that
	$$|u+v|^p \le 2^p \left(|u|^p + |v|^p\right).$$

\epart{b} Suppose $1\le p,q<\infty$ are (H\"older) conjugate exponents, $\frac{1}{p} + \frac{1}{q} = 1$.  Also suppose $x,y\ge 0$.  Prove \emph{Young's inequality}:
	$$xy \le \frac{x^p}{p} + \frac{y^q}{q}.$$


\medskip
\prob{Extra Credit}  Suppose $1\le p < q < \infty$, and consider $\ell^p$ and $\ell^q$.  Prove that for $f\in \ell^1$,
	$$\|f\|_\infty \le \|f\|_q \le \|f\|_p \le \|f\|_1.$$
Also prove that $\ell^p \subset \ell^q$---this inclusion reverses the direction from \textbf{P10}---and give an example to show that this is a proper inclusion.
